{
	int v = a_val ^ 0x55;
	int t = v & 0x7f;
	int seg = t >> 4;
	int r;

	r = ((t & 0x0f) << 4) | 8;
	if (seg)
		r = (r | 0x100) << (seg - 1);
	return (v & 0x80) ? r : 0 - r;
