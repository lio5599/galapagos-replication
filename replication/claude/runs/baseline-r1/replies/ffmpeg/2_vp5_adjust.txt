static int vp5_adjust(int v, int t)
{
    int neg = v < 0;
    int a = neg ? -v : v;
    int d;
    if (a >= 2 * t)
        a = 0;
    d = a - t;
    if (d < 0)
        d = -d;
    a = t - d;
    return neg ? -a : a;
}

    int a = (v + s1) ^ s1;
    int d, s2;
    if (!(a < 2 * t))
        a = 0;
    d = a - t;

    a = t - d;

{
    int a = v;
    int r;
    if (a < 0)
        a = -a;
    if (a < t)
        r = a;
    else if (a < 2 * t)
        r = 2 * t - a;
    else
        r = t - (t < 0 ? -t : t);
    return (v < 0) ? -r : r;
}

    int a = s1 ? -v : v;
    int lo, hi, d;
    a = (a < 2 * t) ? a : 0;
    lo = (a < t) ? a : t;
    hi = (a < t) ? t : a;
    d = hi - lo;
    a = t - d;
    return s1 ? -a : a;
}
