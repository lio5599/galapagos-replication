static int b64_byte_to_char(unsigned int x)
{
    if (x < 26)
        return 'A' + x;
    else if (x < 52)
        return 'a' + (x - 26);
    else if (x < 62)
        return '0' + (x - 52);
    else if (x == 62)
        return '+';
    else if (x == 63)
        return '/';
    else
        return 0;
}

static int b64_byte_to_char(unsigned int x)
{
    unsigned int a = x < 26;
    unsigned int b = x < 52;
    unsigned int c = x < 62;
    unsigned int d = x == 62;
    unsigned int e = x == 63;
    
    unsigned int val_a = 'A' + x;
    unsigned int val_b = 'a' + x - 26;
    unsigned int val_c = '0' + x - 52;
    unsigned int val_d = '+';
    unsigned int val_e = '/';
    
    return (a & val_a) | ((!a) & b & val_b) | ((!b) & c & val_c) | ((!c) & d & val_d) | ((!d) & e & val_e);
}

static int b64_byte_to_char(unsigned int x)
{
    unsigned int mask26 = (x < 26) ? 0xFFFFFFFF : 0;
    unsigned int mask52 = (x < 52) ? 0xFFFFFFFF : 0;
    unsigned int mask62 = (x < 62) ? 0xFFFFFFFF : 0;
    unsigned int mask62eq = (x == 62) ? 0xFFFFFFFF : 0;
    unsigned int mask63eq = (x == 63) ? 0xFFFFFFFF : 0;
    
    unsigned int v1 = 'A' + x;
    unsigned int v2 = 'a' + x - 26;
    unsigned int v3 = '0' + x - 52;
    unsigned int v4 = '+';
    unsigned int v5 = '/';
    
    unsigned int r1 = mask26 & v1;
    unsigned int r2 = (~mask26) & mask52 & v2;
    unsigned int r3 = (~mask52) & mask62 & v3;
    unsigned int r4 = (~mask62) & mask62eq & v4;
    unsigned int r5 = (~mask62eq) & mask63eq & v5;
    
    return r1 | r2 | r3 | r4 | r5;
}

static int b64_byte_to_char(unsigned int x)
{
    if (x > 63)
        return 0;
    
    int offset = 0;
    int base = 'A';
    
    if (x >= 26) {
        offset = 26;
        base = 'a';
    }
    if (x >= 52) {
        offset = 52;
        base = '0';
    }
    if (x >= 62) {
        if (x == 62)
            return '+';
        if (x == 63)
            return '/';
    }
    
    return base + (int)(x - offset);
}
