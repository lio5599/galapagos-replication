static int fpr_lt(uint64_t x, uint64_t y) {
    int64_t sx = *(int64_t *)&x;
    int64_t sy = *(int64_t *)&y;
    
    int same_sign = ~((sx ^ sy) >> 63) & 1;
    int neg_sign = (sx >> 63) & 1;
    
    int lt_pos = (sx - sy) >> 63 & 1;
    int lt_neg = (sy - sx) >> 63 & 1;
    
    return (same_sign & (neg_sign ? lt_neg : lt_pos)) | (~same_sign & neg_sign);
}

static int fpr_lt(uint64_t x, uint64_t y) {
    int64_t sx = *(int64_t *)&x;
    int64_t sy = *(int64_t *)&y;
    
    int sign_bit = (sx ^ sy) >> 63;
    int neg = (sx >> 63) & 1;
    
    int cmp1 = (sx - sy) >> 63 & 1;
    int cmp2 = (sy - sx) >> 63 & 1;
    
    int same_sign_result = (neg & cmp2) | (~neg & cmp1);
    int diff_sign_result = neg & sign_bit;
    
    return (same_sign_result & ~sign_bit) | diff_sign_result;
}

static int fpr_lt(uint64_t x, uint64_t y) {
    int64_t sx = *(int64_t *)&x;
    int64_t sy = *(int64_t *)&y;
    
    int signs_same = ~((sx ^ sy) >> 63) & 1;
    int is_negative = (sx >> 63) & 1;
    
    int less_pos = (sx - sy) >> 63 & 1;
    int less_neg = (sy - sx) >> 63 & 1;
    
    int result = (signs_same & ((~is_negative & less_pos) | (is_negative & less_neg))) |
                 (~signs_same & is_negative);
    
    return result & 1;
}
