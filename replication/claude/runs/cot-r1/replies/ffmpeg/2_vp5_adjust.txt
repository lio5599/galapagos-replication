static int vp5_adjust(int v, int t)
{
    // Reference: returns sign(v) * (t - |c - t|), c = |v| if |v| < 2t else 0 (sign(0) = +).
    // Plain branching version of the same steps: abs, clamp to 0, distance from t, restore sign.
    // v < 0 replaces the (v >> 31) mask. Negating twice equals the xor/add trick.
    // Negative t gives c = 0 and t - |t| = 2t, as in the reference. Zero t gives 0.
    // Inputs that are UB in the reference (v == INT_MIN, overflowing 2*t) are not covered.
    int a = v < 0 ? -v : v;
    int c = a < 2 * t ? a : 0;
    int d = c - t;
    if (d < 0)
        d = -d;
    int r = t - d;
    return v < 0 ? -r : r;
}

static int vp5_adjust(int v, int t)
{
    // Classify a = |v| into a region: 0 if a <= t, 1 if t < a < 2t, 2 if a >= 2t.
    // Region 0 gives a, region 1 gives 2t - a, region 2 gives t - |t| (0, or 2t if t < 0).
    // Regions 0 and 1 need a < 2t, which forces t > 0 and matches t - |a - t|.
    // When t <= 0 the region is always 2. The sign is restored afterwards.
    int a = v < 0 ? -v : v;
    int region = a >= 2 * t ? 2 : (a > t);
    int r;
    switch (region) {
    case 0: r = a; break;
    case 1: r = 2 * t - a; break;
    default: r = t < 0 ? t + t : 0; break;
    }
    return v < 0 ? -r : r;
}
