{

	data ^= data >> 8;

	data ^= data >> 2;

}

static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0u;
	int count;

	data &= 0x7FFFFFF0u;
	for (count = 0; count < 27; count++) {
		parity ^= (data != 0u) ? 1u : 0u;
		data &= data - 1u;
	}
	return parity;
}

	unsigned int v = data & 0x7FFFFFF0u;

	v = (v & 0x33333333u) + ((v >> 2) & 0x33333333u);

	v = (v * 0x01010101u) >> 24;
	return v & 1u;

static unsigned int iec958_parity(unsigned int data)
{
	unsigned int mask;
	unsigned int parity = 0u;

	}
	return parity;
}
