static uint64_t icbrt64(uint64_t x)
{
    // Reference computes floor(cbrt(x)) shifted left by 12 (2*18/3 == 12).
    // Same digit-by-digit cube root, but driven by an up-counting index i.
    // s = 63 - 3*i covers 63,60,...,0, which is the same 22 steps as the reference.
    // The test (x >> s) >= b is kept, so b << s never overflows.
    // The final r * 1 << 12 equals r << 12. x == 0 gives 0 and the max input matches.
    uint64_t r = 0;
    uint64_t b;
    int i;
    int s;

    for (i = 0; i < 22; i++) {
        s = 63 - 3 * i;
        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r++;
        }
    }
    return r << 12;
}
static uint64_t icbrt64(uint64_t x)
{
    // Same algorithm as a while loop: s starts at 66 and is decremented before use.
    // The values of s seen in the body are 63,60,...,0, which is 22 iterations.
    // b is expanded to 3*r*r + 3*r + 1, which is the same value modulo 2^64.
    // The comparison and subtraction are unchanged, so every input behaves identically.
    // The result is scaled by 4096, which equals the reference shift by 12.
    uint64_t r = 0;
    uint64_t b;
    int s = 66;

    while (s > 0) {
        s -= 3;
        r <<= 1;
        b = 3 * r * r + 3 * r + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r++;
        }
    }
    return r * 4096;
}
static uint64_t icbrt64(uint64_t x)
{
    // Branch-free variant: mask is all ones when (x >> s) >= b, otherwise zero.
    // x -= (b << s) & mask subtracts b << s only when the reference would.
    // r += mask & 1 performs the conditional increment.
    // The loop bounds are identical (s = 63 down to 0 in steps of 3).
    // The return value is r << 12, as in the reference.
    uint64_t r = 0;
    uint64_t b;
    uint64_t mask;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        mask = (uint64_t)0 - (uint64_t)((x >> s) >= b);
        x -= (b << s) & mask;
        r += mask & 1;
    }
    return r << 12;
}

{
    // After r <<= 1 the low bit of r is 0, so adding the condition bit equals OR-ing it.
    // The new r is therefore (r << 1) | cond, where cond = ((x >> s) >= b).
    // b is computed from the doubled r before cond is applied, as in the reference.
    // The subtraction of b << s happens under the same condition, so the steps match.
    // The output is r << 12.
    uint64_t r = 0;
    uint64_t b;
    uint64_t c;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r = r << 1;
        b = 3 * r * (r + 1) + 1;
        c = (x >> s) >= b;
        if (c)
            x -= b << s;
        r = r | c;
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Uses a do-while loop with an unsigned shift counter, run exactly 22 times.
    // s takes the values 63,60,...,0, and the loop stops after s == 0.
    // b is computed as (r * (r + 1)) * 3 + 1, which equals 3*r*(r+1)+1 modulo 2^64.
    // The comparison against x >> s and the subtraction match the reference.
    // The result is r shifted left by 12.
    uint64_t r = 0;
    uint64_t b;
    unsigned int s = 63;

    do {
        r <<= 1;
        b = (r * (r + 1)) * 3 + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r++;
        }
        if (s == 0)
            break;
        s -= 3;
    } while (1);
    return r << 12;
}
static uint64_t icbrt64(uint64_t x)
{
    // The top chunk t = x >> s is compared with b, and the difference is reused.
    // When t >= b, x - (b << s) leaves the low s bits unchanged and sets the high part to t - b.
    // So x = ((t - b) << s) | (x & low_mask) is the same as x -= b << s.
    // low_mask = (1 << s) - 1 is valid because s <= 63.
    // r is doubled and incremented as in the reference, and the result is r << 12.
    uint64_t r = 0;
    uint64_t b;
    uint64_t t;
    uint64_t low;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        t = x >> s;
        if (t >= b) {
            low = x & ((((uint64_t)1) << s) - 1);
            x = ((t - b) << s) | low;
            r++;
        }
    }
    return r << 12;
}
static uint64_t icbrt64(uint64_t x)
{
    // r is doubled by adding it to itself, which equals r <<= 1.
    // The 3*r*(r+1) term is computed as m + m + m, where m = r*r + r.
    // The condition test is inverted: skip when (x >> s) < b, otherwise update.
    // This is logically identical to the reference branch.
    // The loop runs over s = 63..0 in steps of 3, and the result is r << 12.
    uint64_t r = 0;
    uint64_t b;
    uint64_t m;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r = r + r;
        m = r * r + r;
        b = m + m + m + 1;
        if ((x >> s) < b)
            continue;
        x -= b << s;
        r += 1;
    }
    return r << 12;
}
static uint64_t icbrt64(uint64_t x)
{
    // Uses a counter k from 22 down to 1, with shift s = 3*(k-1).
    // That gives s = 63,60,...,0, the same sequence as the reference.
    // The core step (double r, compute b, conditional subtract and increment) is unchanged.
    // The conditional increment is written as r += 1 inside the taken branch.
    // The return is (r * 1) << 12, the reference expression with its constant folded.
    uint64_t r = 0;
    uint64_t b;
    int k;
    int s;

    for (k = 22; k > 0; k--) {
        s = 3 * (k - 1);
        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r += 1;
        }
    }
    return (r * 1) << 12;
}
static uint64_t icbrt64(uint64_t x)
{
    // Ternary form of the reference step: the condition c selects the update amounts.
    // x -= c ? (b << s) : 0 subtracts only when (x >> s) >= b.
    // r = 2*r + (c ? 1 : 0), where b is computed from 2*r before adding c.
    // The iteration range (s = 63 down to 0 in steps of 3) is unchanged.
    // The result is r << 12, the same as r * 1 << (2 * 18 / 3).
    uint64_t r = 0;
    uint64_t b;
    int c;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r = 2 * r;
        b = 3 * r * (r + 1) + 1;
        c = (x >> s) >= b;
        x -= c ? (b << s) : 0;
        r += c ? 1 : 0;
    }
    return r << 12;
}
