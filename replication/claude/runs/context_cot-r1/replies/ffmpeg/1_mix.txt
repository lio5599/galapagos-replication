static int mix(int c0, int c1)
{
    // Reference: blends two 15-bit RGB555 colors 2:1 (c0 weighted twice), per channel /3.
    // Approach: extract each channel with shift-then-mask instead of mask-then-shift.
    // (c>>5)&31 equals (c&0x3E0)>>5 exactly, also for negative c (arithmetic shift).
    // Red keeps signed c>>10 and signed truncating division, as in the reference.
    // No overflow is possible: red/3*1024 lies in [-2^31, 2^31-1024].
    int b = ((c0 & 31) << 1) + (c1 & 31);
    int g = (((c0 >> 5) & 31) << 1) + ((c1 >> 5) & 31);
    int r = (c0 >> 10) * 2 + (c1 >> 10);
    return r / 3 * 1024 + g / 3 * 32 + b / 3;
}
static int mix(int c0, int c1)
{
    // Combine the fields with bitwise OR instead of addition.
    // red/3*1024 has its low 10 bits zero in two's complement, even when negative.
    // green/3 <= 31 occupies bits 5..9 and blue/3 <= 31 occupies bits 0..4.
    // The fields do not overlap, so OR gives the same value as the sum.
    int blue = 2 * (c0 & 0x001F) + (c1 & 0x001F);
    int green = (2 * (c0 & 0x03E0) + (c1 & 0x03E0)) >> 5;
    int red = 2 * (c0 >> 10) + (c1 >> 10);
    return (red / 3 * 1024) | (green / 3 * 32) | (blue / 3);
}

    // This matches C truncation toward zero. -red cannot overflow because
    // red >= -3*2^21. Nonnegative red is divided directly.
    int blue = 2 * (c0 & 0x001F) + (c1 & 0x001F);

    int red = 2 * (c0 >> 10) + (c1 >> 10);
    int rq = red < 0 ? -((-red) / 3) : red / 3;

static int mix(int c0, int c1)
{
    // Subtract the remainder first: red - red%3 is an exact multiple of 3.
    // C guarantees red == (red/3)*3 + red%3, with the remainder taking red's sign,
    // so (red - red%3)/3 equals red/3 for all red, including negatives.
    int blue = 2 * (c0 & 0x001F) + (c1 & 0x001F);
    int green = (2 * (c0 & 0x03E0) + (c1 & 0x03E0)) >> 5;
    int red = 2 * (c0 >> 10) + (c1 >> 10);
    int rq = (red - red % 3) / 3;
    return rq * 1024 + green / 3 * 32 + blue / 3;
}

{

    // The error is about 1e-5*x, well below 1/3, so the floor is exact.
    // Red is computed in signed form and combined into the high field with OR,
    // which is valid because red/3*1024 has its low 10 bits zero.
    int blue = 2 * (c0 & 0x001F) + (c1 & 0x001F);
    int green = (2 * (c0 & 0x03E0) + (c1 & 0x03E0)) >> 5;
    int red = 2 * (c0 >> 10) + (c1 >> 10);
    int low = (((green * 21846) >> 16) << 5) | ((blue * 21846) >> 16);
