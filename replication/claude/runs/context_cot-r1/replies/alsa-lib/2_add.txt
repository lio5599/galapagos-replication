static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Reference: saturating unsigned add, returns min(a + b, 4294967295).
	// Approach: compute the wrapped sum and detect overflow when sum < a.
	// If a + b mathematically equals 4294967295 the sum does not wrap and
	// the result is 4294967295, matching the reference's >= branch.
	// When the sum wraps, the true result exceeds the max, so saturate.
	unsigned int s = a + b;
	if (s < a)
		return 4294967295u;
	return s;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Branchless: mask is all ones when the wrapped sum is below a.
	// Overflow happens exactly when the mathematical sum exceeds the max.
	// OR with all ones yields 4294967295, otherwise mask is 0 and sum stays.
	// The exact-max case does not wrap and already equals 4294967295.
	unsigned int s = a + b;
	unsigned int m = 0u - (unsigned int)(s < a);
	return s | m;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Widen to 64 bits so the sum never wraps, then clamp to the max.
	// Sum up to 8589934590 fits in unsigned long long without overflow.
	// Values above 4294967295 clamp; values at or below are returned as is,
	// so the exact-max case yields 4294967295 like the reference.
	unsigned long long s = (unsigned long long)a + (unsigned long long)b;
	return s > 4294967295ULL ? 4294967295u : (unsigned int)s;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Overflow test b > max - a; max - a never underflows for unsigned a.
	// When b equals max - a the sum is exactly 4294967295, which is what
	// the reference returns via its >= branch, so strict > is fine here.
	if (b > 4294967295u - a)
		return 4294967295u;
	return a + b;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// ~b equals 4294967295 - b, so a > ~b means a + b exceeds the max.
	// Equality a == ~b gives a sum of exactly 4294967295, same as reference.
	// Otherwise the sum fits and is returned unchanged.
	if (a > ~b)
		return 4294967295u;
	return a + b;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Symmetric form of the reference: compare b against max - a.
	// b >= max - a is equivalent to a + b >= max mathematically, which is
	// the same condition as a >= max - b, so results match for all inputs.
	if (b >= 4294967295u - a)
		return 4294967295u;
	return a + b;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Ternary with ~a = max - a: b <= ~a means the sum fits in 32 bits.
	// The sum then is at most 4294967295, equal to the reference result.
	// If b > ~a the true sum exceeds the max, so return 4294967295.
	return (b <= ~a) ? a + b : 4294967295u;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// 64-bit sum; bit 32 is the carry and is 1 only when sum exceeds max.
	// Negating the carry gives an all-ones 64-bit mask on overflow, else 0.
	// OR then truncation yields 4294967295 on overflow, or the exact sum.
	unsigned long long s = (unsigned long long)a + (unsigned long long)b;
	return (unsigned int)(s | (0ULL - (s >> 32)));
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Branchless using comparison with b: wrapped sum < b iff overflow,
	// since addition is commutative and wrap occurs only past the max.
	// Mask of all ones saturates; exact-max sum does not wrap and is kept.
	unsigned int s = a + b;
	unsigned int m = 0u - (unsigned int)(s < b);
	return s | m;
}
static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Carry extracted from the high half of a 64-bit sum, 0 or 1.
	// 0 - carry is all ones on overflow, forcing the result to the max.
	// Without carry the low 32-bit sum is exact, including the max itself.
	unsigned int hi = (unsigned int)(((unsigned long long)a + (unsigned long long)b) >> 32);
	return (a + b) | (0u - hi);
}
