static unsigned int iec958_parity(unsigned int data)
{
	// Reference: returns the parity (XOR) of bits 4..30 of data.
	// Mask bits 4..30 with 0x7FFFFFF0, then XOR-fold halves down to bit 0.
	// Each fold keeps bit 0 equal to the XOR of all bits still in range.
	// Bits 0..3 and bit 31 are excluded by the mask, as in the reference.
	unsigned int x = data & 0x7FFFFFF0u;
	x ^= x >> 16;
	x ^= x >> 8;
	x ^= x >> 4;
	x ^= x >> 2;
	x ^= x >> 1;
	return x & 1u;
}

{
	// Shift left by 1 to drop bit 31, then right by 5 to drop bits 0..3.
	// What remains is original bits 4..30 in positions 0..26.
	// Count the set bits until the value is zero, then return the count's low bit.
	// This counts the same 27 bits as the reference loop.
	unsigned int x = (data << 1) >> 5;
	unsigned int count = 0;
	while (x != 0) {
		count += x & 1u;
		x >>= 1;
	}
	return count & 1u;
}

{

	// Keep only those bits, then multiply by 0x11111111 to sum them into bits 28..31.

	// Bit 28 is then the overall parity of the masked bits 4..30.
	unsigned int x = data & 0x7FFFFFF0u;
	x ^= x >> 1;
	x ^= x >> 2;
	x &= 0x11111111u;
	x *= 0x11111111u;
	return (x >> 28) & 1u;
}

{
	// SWAR population count of the masked bits 4..30, then return its low bit.
	// The count's low bit is the parity, which is what the reference returns.
	// The count is at most 27, so the byte sum in the top byte is exact.
	unsigned int x = data & 0x7FFFFFF0u;
