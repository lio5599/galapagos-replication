    // Otherwise sx < sy, which is correct for non-negatives and gives x's sign for mixed signs.

    uint64_t n = 0 - ((x & y) >> 63);

static int fpr_lt(uint64_t x, uint64_t y) {
    // Flipping the sign bit turns a signed comparison into an unsigned one.
    // For non-negative or mixed signs, the result is [(x^S) < (y^S)], which equals [sx < sy].
    // For both negative, the order is reversed, so the result is [y < x] unsigned.
    // Within the same sign, the unsigned and signed orders agree. Equal inputs give 0.
    uint64_t s = 0x8000000000000000ULL;
    if ((x & y) & s) {
        return y < x;
    }
    return (x ^ s) < (y ^ s);
}

    // Computes the result arithmetically from the sign bits and unsigned compares.
    // d = signs differ. In that case the answer is x's sign bit.
    // Same sign: the answer is [x<y] when non-negative and [y<x] when negative.
    // The terms are combined with masks, so exactly one case contributes. Equal inputs give 0.
    int sx = (int)(x >> 63);
    int d = (int)((x ^ y) >> 63);
    int same = (sx ? (y < x) : (x < y));
    return (d & sx) | ((d ^ 1) & same);
