static int val_seg(int val)
{
	unsigned int b = (unsigned int)(val >> 8) & 0xffu;
	int r = 1;
	while (b > 1) {
		b >>= 1;
		r++;
	}
	return r;
}

static int val_seg(int val)
{

	if (b >= 0x80) return 8;
	if (b >= 0x40) return 7;
	if (b >= 0x20) return 6;
	if (b >= 0x10) return 5;
	if (b >= 0x08) return 4;
	if (b >= 0x04) return 3;
	if (b >= 0x02) return 2;
	return 1;

static int val_seg(int val)
{
	unsigned int b = (unsigned int)(val >> 8) & 0xffu;
	unsigned int mask = 0x80u;
	int r = 8;
	while (mask > 1u && !(b & mask)) {
		mask >>= 1;
		r--;
	}
	return r;
}
