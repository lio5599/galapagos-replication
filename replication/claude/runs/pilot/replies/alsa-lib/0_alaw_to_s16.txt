{

	int low = x & 0x0f;
	int seg = (x & 0x70) >> 4;
	int t;

	t = (low << 1) | 1;
	if (seg != 0) {
		t += 0x20;
		t <<= seg - 1;
	}
	t <<= 3;
	return (x & 0x80) ? t : 0 - t;
