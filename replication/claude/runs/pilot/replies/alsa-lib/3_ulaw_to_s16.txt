static int ulaw_to_s16(unsigned char u_val)
{
	unsigned int u = (unsigned char)~u_val;
	int base = ((int)(u & 0x0f) << 3) + 0x84;
	int t;

	switch ((u >> 4) & 0x07) {
	case 0: t = base; break;
	case 1: t = base * 2; break;
	case 2: t = base * 4; break;
	case 3: t = base * 8; break;
	case 4: t = base * 16; break;
	case 5: t = base * 32; break;
	case 6: t = base * 64; break;
	default: t = base * 128; break;
	}

	return (u & 0x80) ? (0x84 - t) : (t - 0x84);
}
