    unsigned int upper, lower, digit, dash, under;

    lower = ((x - 52U) >> 8) & 0xFFU;
    digit = ((x - 62U) >> 8) & 0xFFU;
    dash  = (((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU;
    under = (((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU;

    digit &= lower ^ 0xFFU;
    lower &= upper ^ 0xFFU;

    upper &= x + 'A';
    lower &= x + ('a' - 26);
    digit &= x + ('0' - 52);
    dash  &= '-';
    under &= '_';

    return (int) (upper | lower | digit | dash | under);

{
    unsigned int v = x;
    unsigned int lt26 = ((v - 26U) >> 8) & 0xFFU;
    unsigned int lt52 = ((v - 52U) >> 8) & 0xFFU;
    unsigned int lt62 = ((v - 62U) >> 8) & 0xFFU;
    unsigned int is62 = ~((0U - (v ^ 62U)) >> 8) & 0xFFU;
    unsigned int is63 = ~((0U - (v ^ 63U)) >> 8) & 0xFFU;
    unsigned int part1 = lt26 & (v + (unsigned int) 'A');
    unsigned int part2 = (~lt26 & 0xFFU) & lt52 & (v + (unsigned int) 'a' - 26U);
    unsigned int part3 = (~lt52 & 0xFFU) & lt62 & (v + (unsigned int) '0' - 52U);
    unsigned int part4 = is62 & (unsigned int) '-';
    unsigned int part5 = is63 & (unsigned int) '_';

    return (int) (part5 | part4 | part3 | part2 | part1);
}
