static int ctz(uint64_t in)
{
    int r = 0;

    if (in == 0)
        return 64;

    while ((in & 1) == 0)
    {
        in >>= 1;
        r++;
    }

    return r;
}

static int ctz(uint64_t in)
{
    int r = 0;

    if (in == 0)
        return 64;

    if ((in & 0xFFFFFFFFULL) == 0) { r += 32; in >>= 32; }
    if ((in & 0xFFFFULL) == 0)     { r += 16; in >>= 16; }
    if ((in & 0xFFULL) == 0)       { r += 8;  in >>= 8;  }
    if ((in & 0xFULL) == 0)        { r += 4;  in >>= 4;  }
    if ((in & 0x3ULL) == 0)        { r += 2;  in >>= 2;  }
    if ((in & 0x1ULL) == 0)        { r += 1; }

    return r;
}

static int ctz(uint64_t in)
{
    uint64_t x = in;
    int r = 0, s;

    s = ((x & 0xFFFFFFFFULL) == 0) << 5; r += s; x >>= s;
    s = ((x & 0xFFFFULL) == 0) << 4;     r += s; x >>= s;
    s = ((x & 0xFFULL) == 0) << 3;       r += s; x >>= s;
    s = ((x & 0xFULL) == 0) << 2;        r += s; x >>= s;
    s = ((x & 0x3ULL) == 0) << 1;        r += s; x >>= s;
    s = ((x & 0x1ULL) == 0);             r += s; x >>= s;
    r += (x == 0);

    return r;
}

static int ctz(uint64_t in)

    int i, r = 0;

    for (i = 0; i < 64; i++)
    {
        uint64_t mask = (2ULL << i) - 1;
        r += ((in & mask) == 0);

static int ctz(uint64_t in)
{
    int r = 0;

    if (in == 0)
        return 64;

    while ((in & 0xFFULL) == 0)
    {
        in >>= 8;
        r += 8;
    }

    while ((in & 1ULL) == 0)
    {
        in >>= 1;
        r++;
    }

    return r;
}
