static int weight(int i, int blen, int offset)
{
    int d;
    if (i < 2*offset)
        d = i;
    else if (i > blen-1 - 2*offset)
        d = blen-1 - i;
    else
        return 8;
    if (offset == 1)
        return d ? 5 : 3;
    return 1 + (6*d + offset - 1) / (2*offset - 1);
}

static int weight(int i, int blen, int offset)
{
    int lim = 2*offset;
    int hi = blen - 1 - lim;
    int d;
    if (!(i < lim)) {
        if (!(i > hi))
            return 8;
        d = blen - 1 - i;
    } else {
        d = i;
    }
    if (offset != 1) {
        int num = 6*d + offset - 1;
        int den = 2*offset - 1;
        return 1 + num / den;
    }
    return d != 0 ? 5 : 3;
}

        if (i < 2*offset)
            d = i;

        else
            break;

static int weight(int i, int blen, int offset)
{
    int inLow = i < 2*offset;
    int inHigh = !inLow && (i > blen-1 - 2*offset);
    int d;
    if (!inLow && !inHigh)
        return 8;
    d = inLow ? i : (blen-1 - i);
    if (offset == 1)
        return 3 + 2 * (d != 0);
    return 1 + (6*d + offset - 1) / (2*offset - 1);
}

static int weight(int i, int blen, int offset)
{
    int d, w;
    if (i < 2*offset) {
        d = i;
    } else if (i > blen-1 - 2*offset) {
        d = blen-1 - i;
    } else {
        w = 8;
        return w;
    }
    if (offset == 1) {
        w = 3;
        if (d)
            w += 2;
    } else {
        w = 6*d;
        w += offset - 1;
        w /= 2*offset - 1;
        w += 1;
    }
    return w;
}
