static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    int count = 16 + ch * ((bps + 14) / 8);
    if (ch == 2)
        count += ((2 * bps + 1) * blocksize + 7) / 8;
    else
        count += (ch * bps * blocksize + 7) / 8;
    return count + 2;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    int header = 16;
    int subframe_headers = ch * ((7 + bps + 7) / 8);
    int bits_per_sample_total = (ch == 2) ? (2 * bps + 1) : (ch * bps);
    int data = (bits_per_sample_total * blocksize + 7) / 8;
    int footer = 2;
    return header + subframe_headers + data + footer;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    int per_ch = (bps + 14) / 8;
    int sample_bits;
    int data_bytes;

    switch (ch) {
    case 2:
        sample_bits = 2 * bps + 1;
        break;
    default:
        sample_bits = ch * bps;
        break;
    }
    data_bytes = (sample_bits * blocksize + 7) / 8;
    return 16 + ch * per_ch + data_bytes + 2;
}

    const int frame_header = 16;
    const int frame_footer = 2;
    int stereo = (ch == 2);
    int sample_bits = stereo ? (2 * bps + 1) : (ch * bps);
    int total = frame_header + frame_footer;

    total += ch * ((7 + bps + 7) / 8);
    total += (sample_bits * blocksize + 7) / 8;
    return total;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    int sub = (bps + 14) / 8;
    if (ch == 2) {
        int stereo_bits = (2 * bps + 1) * blocksize;
        return 16 + 2 * sub + (stereo_bits + 7) / 8 + 2;
    } else {
        int bits = ch * bps * blocksize;
        return 16 + ch * sub + (bits + 7) / 8 + 2;
    }
}
