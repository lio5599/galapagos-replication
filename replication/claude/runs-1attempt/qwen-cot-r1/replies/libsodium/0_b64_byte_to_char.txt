static int b64_byte_to_char(unsigned int x)
// Uses bitwise masking similar to reference but with clearer structure
// Computes mask for each range using subtraction and shift
// Range 0-25: mask1 is 0, others are 0xFF
// Range 26-51: mask1 is 0xFF, mask2 is 0, etc.
// Combines results with OR since ranges are mutually exclusive
{
    unsigned int m1 = ((x - 26) >> 8) & 0xFF;
    unsigned int m2 = ((x - 52) >> 8) & 0xFF;
    unsigned int m3 = ((x - 62) >> 8) & 0xFF;
    unsigned int eq62 = ((0U - (x ^ 62)) >> 8) & 0xFF;
    unsigned int eq63 = ((0U - (x ^ 63)) >> 8) & 0xFF;
    return (~m1 & (x + 'A')) |
           (m1 & ~m2 & (x + 'a' - 26)) |
           (m2 & ~m3 & (x + '0' - 52)) |
           (~eq62 & '+') |
           (~eq63 & '/');
}
