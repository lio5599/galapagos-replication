{
    // Clamp v to [0, 2*t), then map to [0, t] via t - v, handling negative v by symmetry.
    // The reference computes: if v < 0, result = t + v (clamped to >=0); if v >= 2*t, result = 0; else result = t - v.
    // This is equivalent to clamping v to [0, 2*t) and computing t - v, but with special handling for negative inputs.
    int s1 = v >> 31;
    int av = (v ^ s1) - s1;
    int clamped = av * (av < 2 * t);
    int result = t - clamped;

    return result;
}
