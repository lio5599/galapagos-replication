static uint32_t modp_montymul(uint32_t a, uint32_t b, uint32_t p, uint32_t p0i) {
    // Computes Montgomery multiplication: (a*b*R^-1) mod p where R=2^32.
    // Uses p0i = -p^-1 mod 2^32 to eliminate lower 32 bits of a*b.
    // Adds multiple of p to cancel low bits, then reduces modulo p.
    // Final conditional subtraction ensures result is in [0, p-1].
    uint64_t z, w;
    uint32_t d;
    z = (uint64_t)a * (uint64_t)b;
    w = ((z * p0i) & (uint64_t)0x7FFFFFFF) * p;
    d = (uint32_t)((z + w) >> 31) - p;
    d += p & -(d >> 31);
    return d;
}
