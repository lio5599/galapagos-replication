{
    // Computes integer cube root of x, scaled by 2^12 (since 2*18/3=12).
    // Uses digit-by-digit restoration algorithm in base 2^3.
    // For each group of 3 bits from MSB, determines next bit of root.
    // Edge case: x=0 returns 0. Large x handled by 64-bit arithmetic.
    uint64_t r = 0;
    uint64_t b;
    int s;
    for (s = 63; s >= 0; s -= 3) {
        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r++;
        }
    }
    return r * 1 << (2 * 18 / 3);
}
