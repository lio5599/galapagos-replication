static int int16_t_negative_mask_1 (int16_t x)
{
  // Reference: returns -1 if x < 0, else 0.
  // Shifts sign bit to LSB via unsigned right shift, then negates.
  // For x < 0, sign bit is 1, u becomes 1, return -1.
  // For x >= 0, sign bit is 0, u becomes 0, return 0.
  // Edge cases: INT16_MIN (-32768) has sign bit 1, returns -1.
  // INT16_MAX (32767) has sign bit 0, returns 0.
  uint16_t u = x;
  u >>= 15;
  return -(int) u;
}

  // Returns -1 if x < 0, else 0.
  // Use arithmetic right shift on signed int16_t to replicate sign bit.
  // Then cast to int and return. For negative x, shift yields -1.
  // For non-negative x, shift yields 0. Direct return works.
  // Edge cases handled correctly by two's complement behavior.
  int16_t s = x >> 15;
  return (int) s;
}

  // Simple and clear, handles all edge cases correctly.
