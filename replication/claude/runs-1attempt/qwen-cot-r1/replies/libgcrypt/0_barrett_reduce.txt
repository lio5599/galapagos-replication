  // Barrett reduction mod 3329 using precomputed v = floor(2^26/3329)
  // Computes a mod 3329 for a in [-32768, 32767]
  // t = round(v*a / 2^26) approximates a/3329, then t*3329 is subtracted
  // The rounding via (1<<25) ensures correct quotient for negative a too
  int16_t t;
  const int16_t v = ((1<<26) + 3329/2)/3329;
  t = ((int32_t)v*a + (1<<25)) >> 26;
  t *= 3329;
  return a - t;
}
