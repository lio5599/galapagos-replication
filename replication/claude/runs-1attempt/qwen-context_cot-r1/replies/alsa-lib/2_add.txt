static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Reference computes saturating addition for 32-bit unsigned integers.
	// If a + b would exceed 4294967295, return 4294967295.
	// Otherwise, return a + b.
	// The check a >= 4294967295 - b detects overflow since 4294967295 is UINT_MAX.
	if (a >= 4294967295 - b)
		return 4294967295;
	return a + b;
}

static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Use subtraction to check for overflow.
	// If a > UINT_MAX - b, then a + b overflows.
	// UINT_MAX is 4294967295 for 32-bit unsigned.
	if (a > 4294967295 - b)
		return 4294967295;
	return a + b;
}

static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Check overflow by verifying if a + b < a (unsigned wraparound).
	// If overflow occurs, a + b wraps to a value less than a.
	// Return UINT_MAX (4294967295) on overflow, else a + b.
	unsigned int sum = a + b;
	if (sum < a)
		return 4294967295;
	return sum;
}

static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Alternative overflow check: if a + b < b, overflow occurred.
	// This is equivalent to sum < a when b > 0.
	// When b == 0, sum == a, so no overflow.
	unsigned int sum = a + b;
	if (sum < b && b != 0)
		return 4294967295;
	return sum;
}

static inline unsigned int add(unsigned int a, unsigned int b)
{
	// Use conditional expression instead of if statement.
	// Check if a >= UINT_MAX - b to detect overflow.
	// Return UINT_MAX on overflow, otherwise a + b.
	return (a >= 4294967295 - b) ? 4294967295 : a + b;
}

	// If sum < a, then overflow happened since adding b >= 0 should not decrease result.

	unsigned int sum = a + b;

	unsigned int sum = a + b;
