int16_t barrett_reduce(int16_t a) {
  // Barrett reduction modulo 3329 using precomputed constant v = floor(2^26 / 3329)
  // Computes q = floor((v*a + 2^25) / 2^26), then returns a - q*3329
  // This is equivalent to a mod 3329 for inputs in int16_t range
  int16_t t;
  const int16_t v = ((1<<26) + 3329/2)/3329;
  t = ((int32_t)v*a + (1<<25)) >> 26;
  t *= 3329;
  return a - t;
}
