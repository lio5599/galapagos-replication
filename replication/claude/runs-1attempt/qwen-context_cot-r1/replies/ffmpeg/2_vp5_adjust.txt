static int vp5_adjust_1(int v, int t)
// Computes clamped absolute value adjustment: returns t - |v| if |v| < t, else 0.
// Uses conditional logic to replicate the bit-twiddling absolute value and clamp.
// Handles negative v by taking absolute value, then clamping against t.
// Equivalent to: if |v| >= t return 0, else return t - |v|.
{
    int abs_v = (v < 0) ? -v : v;
    if (abs_v >= t) {
        return 0;
    }
    return t - abs_v;
}

static int vp5_adjust_2(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Uses ternary operators to compute absolute value and conditional result.
// Directly expresses the mathematical definition of the function.
// Edge case: when v is INT_MIN, abs_v may overflow but original also has UB.
{
    int abs_v = (v < 0) ? -v : v;
    return (abs_v < t) ? (t - abs_v) : 0;
}

static int vp5_adjust_3(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Uses if-else branches for clarity and direct mapping to the logic.
// First computes absolute value, then checks bound and returns appropriate value.
// Maintains same behavior for all defined inputs as reference.
{
    int abs_v;
    if (v < 0) {
        abs_v = -v;
    } else {
        abs_v = v;
    }
    if (abs_v >= t) {
        return 0;
    }
    return t - abs_v;
}

static int vp5_adjust_4(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Uses arithmetic to compute absolute value without branching for sign.
// Then uses conditional to clamp the result.
// Equivalent because v < 0 ? -v : v equals the bit-twiddling abs in reference.
{
    int sign = (v >> 31) & 1;
    int abs_v = (v ^ (sign ? -1 : 0)) - (sign ? -1 : 0);
    if (abs_v >= t) {
        return 0;
    }
    return t - abs_v;
}

static int vp5_adjust_5(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Computes absolute value using arithmetic trick similar to reference.
// Then applies the clamping logic explicitly.
// The bit manipulation for abs is equivalent to the reference's approach.
{
    int s1 = v >> 31;
    int abs_v = (v ^ s1) - s1;
    if (abs_v >= t) {
        return 0;
    }
    return t - abs_v;
}

static int vp5_adjust_6(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Uses a single return statement with nested ternary operators.
// Computes absolute value inline and checks the bound condition.
// Logically identical to reference but expressed more concisely.
{
    int abs_v = (v < 0) ? -v : v;
    return (abs_v < t) ? (t - abs_v) : 0;
}

static int vp5_adjust_7(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Separates the absolute value computation and the clamping into distinct steps.
// Uses standard conditional for absolute value for clarity.
// Produces identical results to reference for all defined inputs.
{
    int abs_v;
    if (v >= 0) {
        abs_v = v;
    } else {
        abs_v = -v;
    }
    if (t <= abs_v) {
        return 0;
    }
    return t - abs_v;
}

static int vp5_adjust_8(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Uses the same bit-twiddling absolute value as the reference.
// Then explicitly checks if the absolute value is within bounds.
// The clamping is done via conditional return for clarity.
{
    int s1 = v >> 31;
    int abs_v = (v ^ s1) - s1;
    if (abs_v < t) {
        return t - abs_v;
    }
    return 0;
}

static int vp5_adjust_9(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Computes absolute value using arithmetic without branches.
// Uses comparison to determine if result should be clamped to zero.
// Equivalent to reference because the abs computation matches.
{
    int abs_v = (v < 0) ? -v : v;
    int result = t - abs_v;
    if (result < 0) {
        return 0;
    }
    return result;
}

static int vp5_adjust_10(int v, int t)
// Computes t - |v| when |v| < t, otherwise 0.
// Computes absolute value, then computes tentative result t - abs_v.
// If tentative result is negative, returns 0, else returns the result.
// This captures the same clamping behavior as the reference implementation.
{
    int abs_v = (v < 0) ? -v : v;
    int diff = t - abs_v;
    return (diff > 0) ? diff : 0;
}
