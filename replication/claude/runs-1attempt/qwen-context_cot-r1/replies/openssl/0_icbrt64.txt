static uint64_t icbrt64(uint64_t x)
{
    // Computes integer cube root of x, then scales by 2^(12) via 2*(2*18/3)=12 shift
    // Uses digit-by-digit method in base 2^3, accumulating result in r
    // Final return: r << 12 since 2*18/3 = 12
    uint64_t r = 0;
    uint64_t b;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r++;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Same algorithm: iteratively determine cube root bits from MSB to LSB
    // Loop processes 3 bits at a time, updating remainder x and root r
    // Final scaling by 2^12 applied at end
    uint64_t r = 0;
    uint64_t b;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r = r * 2;
        b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x = x - (b << s);
            r = r + 1;
        }
    }
    return r * 4096;
}

static uint64_t icbrt64(uint64_t x)
{
    // Digit-by-digit cube root extraction in binary, processing 3 bits per iteration
    // b represents the trial divisor for the next bit of the root
    // Result scaled by 2^12 at the end
    uint64_t r = 0;
    uint64_t b;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        r <<= 1;
        b = 1 + 3 * r * (r + 1);
        if ((x >> s) >= b) {
            x -= (b << s);
            r |= 1;
        }
    }
    return r << 12;
}

    uint64_t r = 0;
