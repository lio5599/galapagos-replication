static int vp5_adjust(int v, int t)
{
    int s1 = v >> 31;
    v ^= s1;
    v -= s1;
    v *= (v < 2 * t);
    v -= t;
    int s2 = v >> 31;
    v ^= s2;
    v -= s2;
    v = t - v;
    v += s1;
    v ^= s1;
    return v;
}
