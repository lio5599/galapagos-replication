```c
static int ctz(uint64_t in)
{
    int r = 0;
    while (r < 64 && !(in & 1)) {
        in >>= 1;
        r++;
    }
    return r;
}
```

```c
static int ctz(uint64_t in)
{
    int r = 0;
    for (; r < 64; r++) {
        if (in & 1) break;
        in >>= 1;
    }
    return r;
}
```
