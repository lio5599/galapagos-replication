    int64_t sx = *(int64_t *)&x;
    int64_t sy = *(int64_t *)&y;
    
    int result;
    
    if (sx >= 0) {
        if (sy >= 0) {
            result = sx < sy;
        } else {
            result = 0;
        }
    } else {
        if (sy >= 0) {
            result = 1;
        } else {
            result = sx > sy;
        }
    }
    
    return result;
}
