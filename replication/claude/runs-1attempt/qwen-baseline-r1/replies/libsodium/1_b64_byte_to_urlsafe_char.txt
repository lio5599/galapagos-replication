static int b64_byte_to_urlsafe_char(unsigned int x)
{
    if (x < 26) return x + 'A';
    if (x < 52) return x + 'a' - 26;
    if (x < 62) return x + '0' - 52;
    if (x == 62) return '-';
    if (x == 63) return '_';
    return 0;
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    if (x > 63) return 0;
    if (x < 26) return 'A' + x;
    if (x < 52) return 'a' + (x - 26);
    if (x < 62) return '0' + (x - 52);
    return (x == 62) ? '-' : '_';
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    unsigned int mask = (x < 26);
    int result = (mask & (x + 'A')) |
                 (~mask & ((x < 52) & (x + 'a' - 26))) |
                 (~(mask | (x < 52)) & ((x < 62) & (x + '0' - 52))) |
                 (~(mask | (x < 52) | (x < 62)) & ((x == 62) & '-')) |
                 (~(mask | (x < 52) | (x < 62) | (x == 62)) & ((x == 63) & '_'));
    return result;
}
