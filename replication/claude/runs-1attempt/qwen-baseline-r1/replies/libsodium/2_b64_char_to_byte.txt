```c
static unsigned int b64_char_to_byte(int c)
{
    unsigned int result = 0;
    if (c >= 'A' && c <= 'Z')
        result = c - 'A';
    else if (c >= 'a' && c <= 'z')
        result = c - 'a' + 26;
    else if (c >= '0' && c <= '9')
        result = c - '0' + 52;
    else if (c == '+')
        result = 62;
    else if (c == '/')
        result = 63;
    else
        result = 0;
    return result;
}
```

```c
    int is_slash = (c == '/');

    if (in_upper)
        val = c - 'A';

    else if (in_digit)
        val = c - '0' + 52;
    else if (is_plus)
        val = 62;
    else if (is_slash)
        val = 63;

    return val;
}
```

```c
static unsigned int b64_char_to_byte(int c)
{
    unsigned int result = 0;
    int upper = (c >= 'A' && c <= 'Z');
    int lower = (c >= 'a' && c <= 'z');
    int digit = (c >= '0' && c <= '9');
    int plus = (c == '+');
    int slash = (c == '/');

    result = upper * (c - 'A') +
             lower * (c - 'a' + 26) +
             digit * (c - '0' + 52) +
             plus * 62 +
             slash * 63;

    return result;
}
```

```c
static unsigned int b64_char_to_byte(int c)

    if (c >= 'A' && c <= 'Z')
        return (unsigned int)(c - 'A');

        return (unsigned int)(c - 'a' + 26);
    if (c >= '0' && c <= '9')
        return (unsigned int)(c - '0' + 52);
    switch (c) {

        case '/': return 63;

    }
```
