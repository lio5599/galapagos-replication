static int b64_byte_to_char(unsigned int x)
{
    if (x < 26)
        return 'A' + x;
    else if (x < 52)
        return 'a' + (x - 26);
    else if (x < 62)
        return '0' + (x - 52);
    else if (x == 62)
        return '+';
    else if (x == 63)
        return '/';
    else
        return 0;
}

    if (x < 26)
        return x + 'A';
    x -= 26;
    if (x < 26)
        return x + 'a';
    x -= 26;
    if (x < 10)
        return x + '0';
    x -= 10;
    if (x == 0)
        return '+';
    if (x == 1)
        return '/';
    return 0;
