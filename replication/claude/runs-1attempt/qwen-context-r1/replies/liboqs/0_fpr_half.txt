static uint64_t fpr_half(uint64_t x) {
    uint64_t exp = (x >> 52) & 0x7FF;
    uint64_t mantissa = x & ((1ULL << 52) - 1);
    uint64_t sign = x & (1ULL << 63);
    
    if (exp == 0) {
        return x;
    }
    
    exp -= 1;
    return sign | (exp << 52) | mantissa;
}
