static int fpr_lt(uint64_t x, uint64_t y) {
    int64_t sx = *(int64_t *)&x;
    int64_t sy = *(int64_t *)&y;
    int sx_sign = sx >> 63;
    int sy_sign = sy >> 63;
    int signs_same = !(sx_sign ^ sy_sign);
    
    if (signs_same) {
        if (sx_sign == 0) {
            return sx < sy;
        } else {
            return sx > sy;
        }
    } else {
        return sx_sign < sy_sign;
    }
}

static int fpr_lt(uint64_t x, uint64_t y) {
    int64_t sx = *(int64_t *)&x;
    int64_t sy = *(int64_t *)&y;
    int sx_neg = (int)(sx >> 63);
    int sy_neg = (int)(sy >> 63);
    int diff_sign = sx_neg ^ sy_neg;
    
    if (diff_sign) {
        return sx_neg;
    } else {
        return (sx - sy) >> 63 & 1;
    }
}
