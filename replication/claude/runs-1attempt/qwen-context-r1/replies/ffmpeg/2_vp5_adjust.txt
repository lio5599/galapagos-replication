static int vp5_adjust(int v, int t)
{
    int s1 = v >> 31;
    v ^= s1;
    v -= s1;
    int cond = v < 2 * t;
    v = cond ? v : 0;
    v -= t;
    int s2 = v >> 31;
    v ^= s2;
    v -= s2;
    v = t - v;
    v += s1;
    v ^= s1;
    return v;
}

static int vp5_adjust(int v, int t)
{
    int abs_v = (v < 0) ? -v : v;
    int sign = (v < 0) ? -1 : 0;
    int clamped = (abs_v < 2 * t) ? abs_v : 0;
    int diff = clamped - t;
    int abs_diff = (diff < 0) ? -diff : diff;
    int result = t - abs_diff;
    result += sign;
    result ^= sign;
    return result;
}

    int bounded = av < 2 * t ? av : 0;
    int delta = bounded - t;

static int vp5_adjust(int v, int t)
{
    int abs_v;
    int sign_v;
    if (v >= 0) {
        abs_v = v;
        sign_v = 0;
    } else {
        abs_v = -v;
        sign_v = -1;
    }
    int capped = abs_v < 2 * t ? abs_v : 0;
    int diff = capped - t;
    int abs_diff;
    if (diff >= 0) {
        abs_diff = diff;
    } else {
        abs_diff = -diff;
    }
    int res = t - abs_diff;
    if (sign_v != 0) {
        res = -res;
    }
    return res;
}
