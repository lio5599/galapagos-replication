static int b64_byte_to_char(unsigned int x)
{
    if (x < 26)
        return x + 'A';
    else if (x < 52)
        return x + 'a' - 26;
    else if (x < 62)
        return x + '0' - 52;
    else if (x == 62)
        return '+';
    else if (x == 63)
        return '/';
    else
        return 0;
}

        return y + 'A';

    if (y < 10)

static int b64_byte_to_char(unsigned int x)
{
    unsigned int mask1 = ((x - 26) >> 8) & 0xFF;
    unsigned int mask2 = ((x - 52) >> 8) & 0xFF;
    unsigned int mask3 = ((x - 62) >> 8) & 0xFF;
    unsigned int eq62 = ((0U - (x ^ 62)) >> 8) & 0xFF;
    unsigned int eq63 = ((0U - (x ^ 63)) >> 8) & 0xFF;
    
    int r1 = (mask1 & (x + 'A'));
    int r2 = ((~mask1 & mask2) & (x + 'a' - 26));
    int r3 = ((~mask2 & mask3) & (x + '0' - 52));
    int r4 = ((~eq62) & '+');
    int r5 = ((~eq63) & '/');
    
    return r1 | r2 | r3 | r4 | r5;
}
