static unsigned int b64_char_to_byte(int c)
{
    unsigned int result = 0;
    if (c >= 'A' && c <= 'Z') {
        result = c - 'A';
    } else if (c >= 'a' && c <= 'z') {
        result = c - 'a' + 26;
    } else if (c >= '0' && c <= '9') {
        result = c - '0' + 52;
    } else if (c == '+') {
        result = 62;
    } else if (c == '/') {
        result = 63;
    }
    return result;
}
