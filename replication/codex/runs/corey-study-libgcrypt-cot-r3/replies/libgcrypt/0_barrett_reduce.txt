int16_t barrett_reduce(int16_t a) {
  // The reference computes a modulo 3329 centered in [-1664, 1664].
  // Its rounded reciprocal selects the nearest quotient for every int16_t.
  // C remainder has magnitude below 3329; one correction centers it.
  // Wide arithmetic handles both int16_t endpoints without overflow.
  int32_t r = (int32_t)a % 3329;
  if (r > 1664)
    r -= 3329;
  else if (r < -1664)
    r += 3329;
  return (int16_t)r;
}

  int32_t x = a;

}

  // Repeatedly add or subtract the modulus until the value is centered.
  // Each step preserves congruence and moves toward [-1664, 1664].
  // The unique centered result is exactly the reference result.
  // At most ten iterations are needed, including at either input endpoint.
  int32_t r = a;
  while (r > 1664)
    r -= 3329;
  while (r < -1664)
    r += 3329;
  return (int16_t)r;
}

  int32_t r = (int32_t)a + 34954;

int16_t barrett_reduce(int16_t a) {
  // Process all seventeen bits of the positive biased input by long division.
  // Each step doubles a remainder and adds a bit, needing at most one subtraction.
  // The invariant keeps r in [0, 3328] with no overflowing intermediates.
  // Removing the 1664 bias yields the reference result for all int16_t inputs.
  int32_t x = (int32_t)a + 34954;
  int32_t r = 0;
  int bit;
  for (bit = 16; bit >= 0; --bit) {
    r = r * 2 + ((x >> bit) & 1);
    if (r >= 3329)
      r -= 3329;
  }
  return (int16_t)(r - 1664);
}

  // Use the reference reciprocal, 20159, but replace its arithmetic shift.
  // Division truncates toward zero; correcting negative remainders gives floor.
  // This reproduces the reference's rounded quotient without a negative shift.
  // Wide subtraction directly produces the centered result at both endpoints.
  int32_t x = (int32_t)20159 * a + 33554432;
  int32_t q = x / 67108864;
  if (x % 67108864 < 0)
    --q;

}

  // Since 20159 * 3329 equals 67108864 + 447, quotient error is below 0.000066.

  // Thus the quotient is nearest and centered reduction is odd for every input.
