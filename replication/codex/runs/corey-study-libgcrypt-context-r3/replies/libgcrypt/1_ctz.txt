#include <stdint.h>

static int ctz(uint64_t in)
{
    int r = 0;
    if (in == 0)
        return 64;

    while ((in & 1) == 0) {
        ++r;
        in >>= 1;
    }
    return r;
}

static int ctz(uint64_t in)
{
    int r = 0;

    if (in == 0)
        return 64;
    if ((in & UINT64_C(0xffffffff)) == 0) {
        r += 32;
        in >>= 32;
    }
    if ((in & UINT64_C(0xffff)) == 0) {
        r += 16;
        in >>= 16;
    }
    if ((in & UINT64_C(0xff)) == 0) {
        r += 8;
        in >>= 8;
    }
    if ((in & UINT64_C(0xf)) == 0) {
        r += 4;
        in >>= 4;
    }
    if ((in & UINT64_C(3)) == 0) {
        r += 2;
        in >>= 2;
    }
    return r + ((in & 1) == 0);
}

#include <stdint.h>

static int ctz(uint64_t in)
{
    int r = 0;

    if (in == 0)
        return 64;

    while ((in & UINT64_C(0xf)) == 0) {
        r += 4;
        in >>= 4;
    }

    if ((in & 1) != 0)
        return r;
    if ((in & 2) != 0)
        return r + 1;
    if ((in & 4) != 0)
        return r + 2;
    return r + 3;
}

    int r = 0;
    int i;

    x |= x << 1;
    x |= x << 2;
    x |= x << 4;
    x |= x << 8;
    x |= x << 16;
    x |= x << 32;

    for (i = 0; i < 64; ++i) {
        r += (int)((x & 1) ^ 1);
        x >>= 1;
    }
    return r;
}
