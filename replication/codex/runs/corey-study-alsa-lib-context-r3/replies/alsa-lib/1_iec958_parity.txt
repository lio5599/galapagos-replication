static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	int bit;

	data >>= 4;
	for (bit = 0; bit < 27; ++bit) {
		parity ^= data & 1U;
		data >>= 1;
	}
	return parity;
}

static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	unsigned int remaining = 27;

	data >>= 4;
	do {
		parity += data & 1U;
		data >>= 1;

static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	int pair;

	data >>= 4;
	for (pair = 0; pair < 13; ++pair) {
		parity ^= (data ^ (data >> 1)) & 1U;

	}
