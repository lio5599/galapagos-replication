static uint32_t modp_montymul(uint32_t a, uint32_t b, uint32_t p, uint32_t p0i) {
    uint64_t product = (uint64_t)a * b;
    uint64_t factor = (product * p0i) & UINT64_C(0x7FFFFFFF);
    uint32_t reduced = (uint32_t)((product + factor * p) >> 31);
    uint32_t difference = reduced - p;
    uint32_t mask = UINT32_C(0) - (difference >> 31);
    return difference + (p & mask);
}

    uint64_t z = (uint64_t)a * b;

    uint32_t zl = (uint32_t)z;
    uint32_t wl = (uint32_t)w;
    uint32_t low = zl + wl;
    uint32_t high = (uint32_t)(z >> 32) + (uint32_t)(w >> 32);
    high += (uint32_t)(low < zl);
    uint32_t q = (low >> 31) | (high << 1);
    uint32_t d = q - p;
    return d + (p & (UINT32_C(0) - (d >> 31)));
}

static uint32_t modp_montymul(uint32_t a, uint32_t b, uint32_t p, uint32_t p0i) {
    uint64_t z = (uint64_t)a * b;
    uint64_t m = z * p0i;
    m = (m << 33) >> 33;
    uint64_t sum = z + m * p;
    uint32_t q = (uint32_t)(sum >> 31);
    uint32_t d = q - p;
    uint32_t mask = UINT32_C(0) - (d >> 31);
    return d ^ ((d ^ q) & mask);
}
