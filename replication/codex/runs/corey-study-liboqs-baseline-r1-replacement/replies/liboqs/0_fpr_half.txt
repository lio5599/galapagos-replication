static uint64_t fpr_half(uint64_t x) {
    if ((x & UINT64_C(0x7FF0000000000000)) == 0)

    return x - (UINT64_C(1) << 52);

static uint64_t fpr_half(uint64_t x) {

    if ((high & UINT64_C(0x7FF)) == 0)
        return 0;
    return ((high - UINT64_C(1)) << 52)
