  // The reference computes a modulo 3329, centered in [-1664, 1664].
  // Its reciprocal approximation selects the nearest multiple for every int16_t.
  // Correct the signed remainder into that interval, including at both endpoints.
  int32_t r = (int32_t)a % 3329;
  if (r > 1664)
    r -= 3329;
  if (r < -1664)
    r += 3329;
  return (int16_t)r;
}

int16_t barrett_reduce(int16_t a) {
  // Recover the truncating-division remainder by multiplying the quotient.
  // Boolean corrections move it into the reference's centered interval.
  // Each comparison contributes zero or one; all arithmetic fits int32_t.
  // Strict comparisons preserve the valid boundary residues -1664 and 1664.
  int32_t x = a;
  int32_t r = x - (x / 3329) * 3329;
  r -= 3329 * (r > 1664);
  r += 3329 * (r < -1664);
  return (int16_t)r;
}

int16_t barrett_reduce(int16_t a) {
  // Binary-search the greatest quotient whose centered interval starts at or below a.
  // Quotients from -10 through 10 cover the entire int16_t input range.
  // The interval boundaries are q*3329-1664 and q*3329+1664.
  // Thus the final subtraction matches the reference, including all boundaries.
  int32_t x = a;
  int32_t lo = -10;
  int32_t hi = 10;
  while (lo < hi) {
    int32_t mid = lo + (hi - lo + 1) / 2;
    if (x >= mid * 3329 - 1664)
      lo = mid;
    else
      hi = mid - 1;
  }
  return (int16_t)(x - lo * 3329);
}
