static int alaw_to_s16(unsigned char a_val)
{
	int magnitude;
	int segment;

	a_val ^= 0x55;
	segment = (a_val >> 4) & 7;
	magnitude = ((a_val & 15) << 4) + 8;
	if (segment != 0)
		magnitude = (magnitude + 256) << (segment - 1);

	return (a_val & 128) ? magnitude : -magnitude;
}

{

	int magnitude;

	magnitude = (int)((value & 15u) * 16u + 8u);

	switch ((value >> 4) & 7u) {

	case 1: magnitude += 256; break;
	case 2: magnitude = (magnitude + 256) * 2; break;
	case 3: magnitude = (magnitude + 256) * 4; break;
	case 4: magnitude = (magnitude + 256) * 8; break;
	case 5: magnitude = (magnitude + 256) * 16; break;
	case 6: magnitude = (magnitude + 256) * 32; break;
	case 7: magnitude = (magnitude + 256) * 64; break;
	}

	return (value & 128u) ? magnitude : -magnitude;

static int alaw_to_s16(unsigned char a_val)
{
	static const int base[8] = {
		8, 264, 528, 1056, 2112, 4224, 8448, 16896
	};
	static const int step[8] = {
		16, 16, 32, 64, 128, 256, 512, 1024
	};
	int value;
	int segment;
	int magnitude;

	value = a_val ^ 0x55;
	segment = (value >> 4) & 7;
	magnitude = base[segment] + (value & 15) * step[segment];

	return (value & 128) ? magnitude : -magnitude;
}

static int alaw_to_s16(unsigned char a_val)
{
	int value;
	int segment;
	int magnitude;

	value = a_val ^ 0x55;
	segment = (value >> 4) & 7;
	magnitude = (value & 15) * 16 + 8;

	if (segment > 0) {
		magnitude += 256;
		while (--segment > 0)
			magnitude *= 2;
	}

	if (!(value & 128))
		magnitude = -magnitude;
	return magnitude;
}

static int alaw_to_s16(unsigned char a_val)
{
	int segment;
	int mantissa;
	int magnitude;

	segment = ((a_val >> 4) ^ 5) & 7;
	mantissa = (a_val ^ 5) & 15;

	if (segment == 0)
		magnitude = mantissa * 16 + 8;
	else
		magnitude = (mantissa * 2 + 33) << (segment + 2);

	return (a_val & 128) ? magnitude : -magnitude;
}

static int alaw_to_s16(unsigned char a_val)
{
	int value;
	int segment;
	int magnitude;
	int shift;

	value = a_val ^ 0x55;
	segment = (value >> 4) & 7;
	shift = segment - (segment != 0);
	magnitude = ((value & 15) * 16 + 8
		+ 256 * (segment != 0)) << shift;

	return magnitude * (2 * ((value >> 7) & 1) - 1);
}

static int alaw_to_s16(unsigned char a_val)
{
	int code;
	int magnitude;

	code = (a_val ^ 0x55) & 127;

	if (code < 16)
		magnitude = code * 16 + 8;
	else if (code < 32)
		magnitude = (code - 16) * 16 + 264;
	else if (code < 48)
		magnitude = (code - 32) * 32 + 528;
	else if (code < 64)
		magnitude = (code - 48) * 64 + 1056;
	else if (code < 80)
		magnitude = (code - 64) * 128 + 2112;
	else if (code < 96)
		magnitude = (code - 80) * 256 + 4224;
	else if (code < 112)
		magnitude = (code - 96) * 512 + 8448;
	else
		magnitude = (code - 112) * 1024 + 16896;

	return (a_val & 128) ? magnitude : -magnitude;
}

static int alaw_to_s16(unsigned char a_val)
{
	int value;
	int segment;
	int magnitude;

	value = a_val ^ 0x55;
	segment = (value >> 4) & 7;
	magnitude = ((value & 15) << 1) | 1;

	if (segment == 0)
		magnitude <<= 3;
	else {
		magnitude |= 32;
		magnitude <<= segment + 2;
	}

	return (value & 128) ? magnitude : -magnitude;
}
