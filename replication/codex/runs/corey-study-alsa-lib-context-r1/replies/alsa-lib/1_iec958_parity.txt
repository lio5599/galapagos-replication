static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	int bit;

	data >>= 4;
	for (bit = 0; bit < 27; ++bit) {
		parity ^= data & 1u;
		data >>= 1;
	}
	return parity;
}

static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	int remaining = 27;

	data >>= 4;
	do {
		parity ^= data;
		data >>= 1;
	} while (--remaining);
	return parity & 1u;
}

{
	unsigned int parity = 0;
	int bit;

	data >>= 4;
	for (bit = 0; bit < 27; ++bit) {
		parity = (parity + data % 2u) % 2u;
		data /= 2u;
	}
	return parity;
}
