static int b64_byte_to_char(unsigned int x)
{
    // Maps 0..63 to Base64 characters; other inputs follow byte-mask arithmetic.
    // Name each original mask and combine the same five masked values.
    // Unsigned wraparound and partial masks are preserved for every input.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;

    return (int)((a & (x + 'A')) |
                 ((a ^ 255U) & b & (x + ('a' - 26))) |
                 ((b ^ 255U) & c & (x + ('0' - 52))) |
                 (d & '+') | (e & '/'));
}

static int b64_byte_to_char(unsigned int x)
{
    // Compute subtraction masks using the high byte and a low-byte borrow.
    // Each threshold is below 256, so this reproduces subtraction's next byte.
    // Masking preserves wraparound even at zero and the unsigned maximum.
    // Keep the original equality-like masks, including their partial values.
    unsigned int lo = x & 255U;
    unsigned int hi = x >> 8;
    unsigned int a = (hi - (unsigned int)(lo < 26U)) & 255U;
    unsigned int b = (hi - (unsigned int)(lo < 52U)) & 255U;
    unsigned int c = (hi - (unsigned int)(lo < 62U)) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;

    return (int)((a & (x + 'A')) |
                 (~a & b & (x + ('a' - 26))) |
                 (~b & c & (x + ('0' - 52))) |
                 (d & '+') | (e & '/'));
}

static int b64_byte_to_char(unsigned int x)
{
    // Evaluate the five terms one output bit at a time.
    // Mask bit i comes from subtraction bit i+8, with the same inversions.
    // All arithmetic remains unsigned; no assumption that x is below 64 is made.
    unsigned int a = x - 26U;
    unsigned int b = x - 52U;
    unsigned int c = x - 62U;
    unsigned int d = 0U - (x ^ 62U);
    unsigned int e = 0U - (x ^ 63U);
    unsigned int va = x + 'A';
    unsigned int vb = x + ('a' - 26);
    unsigned int vc = x + ('0' - 52);
    unsigned int result = 0U;
    unsigned int i;

    for (i = 0U; i < 8U; ++i) {
        unsigned int ma = (a >> (i + 8U)) & 1U;
        unsigned int mb = (b >> (i + 8U)) & 1U;
        unsigned int mc = (c >> (i + 8U)) & 1U;
        unsigned int md = ((d >> (i + 8U)) & 1U) ^ 1U;
        unsigned int me = ((e >> (i + 8U)) & 1U) ^ 1U;
        unsigned int bit = (ma & (va >> i)) |
                           ((ma ^ 1U) & mb & (vb >> i)) |
                           ((mb ^ 1U) & mc & (vc >> i)) |
                           (md & ((unsigned int)'+' >> i)) |
                           (me & ((unsigned int)'/' >> i));
        result |= bit << i;
    }
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Store the five selection masks and candidate values in parallel arrays.
    // OR reduction is exactly the reference's five-term expression.
    // Masks remain bytes rather than Boolean tests, preserving out-of-range cases.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int masks[5];
    unsigned int values[5];
    unsigned int result = 0U;
    unsigned int i;

    masks[0] = a;
    masks[1] = (a ^ 255U) & b;
    masks[2] = (b ^ 255U) & c;
    masks[3] = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    masks[4] = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    values[0] = x + 'A';
    values[1] = x + ('a' - 26);
    values[2] = x + ('0' - 52);
    values[3] = '+';
    values[4] = '/';
    for (i = 0U; i < 5U; ++i)
        result |= masks[i] & values[i];
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Apply De Morgan's law to the OR of the five masked terms.
    // Final byte masking makes full-width complements match byte complements.
    // Original unsigned subtractions preserve all boundary and large-input cases.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    unsigned int va = x + 'A';
    unsigned int vb = x + ('a' - 26);
    unsigned int vc = x + ('0' - 52);

    return (int)(255U & ~((~a | ~va) &
                          (a | ~b | ~vb) &
                          (b | ~c | ~vc) &
                          (~d | ~(unsigned int)'+') &
                          (~e | ~(unsigned int)'/')));
}

static int b64_byte_to_char(unsigned int x)
{
    // Test whether each output bit is supplied by any of the five terms.
    // Logical tests apply to individual bits, not to whole partial masks.
    // Thus arbitrary unsigned inputs retain exactly the reference's bit behavior.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    unsigned int va = x + 'A';
    unsigned int vb = x + ('a' - 26);
    unsigned int vc = x + ('0' - 52);
    unsigned int result = 0U;
    unsigned int bit;

    for (bit = 1U; bit < 256U; bit <<= 1) {
        if (((a & bit) && (va & bit)) ||
            (!(a & bit) && (b & bit) && (vb & bit)) ||
            (!(b & bit) && (c & bit) && (vc & bit)) ||
            ((d & bit) && ((unsigned int)'+' & bit)) ||
            ((e & bit) && ((unsigned int)'/' & bit)))
            result |= bit;
    }
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Accumulate terms using r OR t = r XOR (t AND NOT r).
    // Only previously unset bits are toggled, even when terms overlap.
    // The original byte masks and unsigned wraparound cover every input.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    unsigned int result = a & (x + 'A');
    unsigned int term;

    term = (~a & b) & (x + ('a' - 26));
    result ^= term & ~result;
    term = (~b & c) & (x + ('0' - 52));
    result ^= term & ~result;
    term = d & '+';
    result ^= term & ~result;
    term = e & '/';
    result ^= term & ~result;
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Accumulate OR using addition of only bits absent from the running result.
    // Disjoint set bits cannot generate carries, so each addition equals OR.
    // Byte masks bound the result to 255 and preserve all unsigned edge cases.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    unsigned int result = a & (x + 'A');

    result += ((~a & b) & (x + ('a' - 26))) & ~result;
    result += ((~b & c) & (x + ('0' - 52))) & ~result;
    result += (d & '+') & ~result;
    result += (e & '/') & ~result;
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Compute the result as two independent four-bit slices.
    // Bitwise gates do not carry between slices, so recombining reproduces OR.
    // Slice masks retain partial-mask behavior for inputs outside 0..63.
    unsigned int a = (x - 26U) >> 8;
    unsigned int b = (x - 52U) >> 8;
    unsigned int c = (x - 62U) >> 8;
    unsigned int d = ~((0U - (x ^ 62U)) >> 8);
    unsigned int e = ~((0U - (x ^ 63U)) >> 8);
    unsigned int va = x + 'A';
    unsigned int vb = x + ('a' - 26);
    unsigned int vc = x + ('0' - 52);
    unsigned int result = 0U;
    unsigned int shift;

    for (shift = 0U; shift < 8U; shift += 4U) {
        unsigned int ma = (a >> shift) & 15U;
        unsigned int mb = (b >> shift) & 15U;
        unsigned int mc = (c >> shift) & 15U;
        unsigned int md = (d >> shift) & 15U;
        unsigned int me = (e >> shift) & 15U;
        unsigned int part = (ma & (va >> shift)) |
                            ((ma ^ 15U) & mb & (vb >> shift)) |
                            ((mb ^ 15U) & mc & (vc >> shift)) |
                            (md & ((unsigned int)'+' >> shift)) |
                            (me & ((unsigned int)'/' >> shift));
        result |= part << shift;
    }
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Keep masks at full unsigned width and truncate only the final result.
    // AND, OR, and complement act independently on each bit.
    // Consequently higher bits cannot affect the reference's low eight bits.
    // Unsigned arithmetic preserves wraparound for zero and maximum inputs.
    unsigned int a = (x - 26U) >> 8;
    unsigned int b = (x - 52U) >> 8;
    unsigned int c = (x - 62U) >> 8;
    unsigned int d = (0U - (x ^ 62U)) >> 8;
    unsigned int e = (0U - (x ^ 63U)) >> 8;
    unsigned int result;

    result = (a & (x + 'A')) |
             (~a & b & (x + ('a' - 26))) |
             (~b & c & (x + ('0' - 52))) |
             (~d & (unsigned int)'+') |
             (~e & (unsigned int)'/');
    return (int)(result & 255U);
}
