static int ctz(uint64_t in)
{
 // The reference counts trailing zero bits, returning 64 for zero.
 // Shift past each low zero bit until a one appears.
 // Handling zero separately keeps the loop finite for every input.
 int r = 0;
 if (in == 0)
  return 64;
 while ((in & 1) == 0)
 {
  in >>= 1;
  r++;
 }
 return r;
}

static int ctz(uint64_t in)
{
 // Examine bit positions from least to most significant.
 // The first set bit's position equals the trailing-zero count.
 // If no bit is set, all 64 positions are zero.
 int i;
 for (i = 0; i < 64; i++)
 {
  if (((in >> i) & 1) != 0)
   return i;
 }
 return 64;
}

static int ctz(uint64_t in)
{
 // Move a single-bit mask upward until it meets a set bit.
 // Unsigned wraparound makes the mask zero after 64 shifts.
 // Thus zero input returns 64 without any shift by 64.
 uint64_t mask = 1;
 int r = 0;
 while (mask != 0 && (in & mask) == 0)
 {
  r++;
  mask <<= 1;
 }
 return r;
}

static int ctz(uint64_t in)
{
 // Track whether any examined bit has been one.
 // Increment only while all bits examined so far are zero.
 // Zero input increments on all 64 iterations.
 int i;
 int seen = 0;
 int r = 0;
 for (i = 0; i < 64; i++)
 {
  seen |= (int)((in >> i) & 1);
  r += (seen == 0);
 }
 return r;
}

static int ctz(uint64_t in)
{
 // Skip zero low blocks of successively smaller sizes.
 // Each skipped block contributes its size to the result.
 // A separate zero case returns 64; nonzero inputs retain a set bit.
 int r = 0;
 if (in == 0)
  return 64;
 if ((in & 0xFFFFFFFFULL) == 0)
 {
  r += 32;
  in >>= 32;
 }
 if ((in & 0xFFFFULL) == 0)
 {
  r += 16;
  in >>= 16;
 }
 if ((in & 0xFFULL) == 0)
 {
  r += 8;
  in >>= 8;
 }
 if ((in & 0xFULL) == 0)
 {
  r += 4;
  in >>= 4;
 }
 if ((in & 3) == 0)
 {
  r += 2;
  in >>= 2;
 }
 r += ((in & 1) == 0);
 return r;
}

static int ctz(uint64_t in)
{
 // Complementing the input and masking with input minus one
 // produces one set bit for each original trailing zero.
 // For zero, unsigned subtraction produces 64 set bits.
 // Clear one set bit per iteration to count them.
 uint64_t bits = ~in & (in - (uint64_t)1);
 int r = 0;
 while (bits != 0)
 {
  bits &= bits - (uint64_t)1;
  r++;
 }
 return r;
}

static int ctz(uint64_t in)
{
 // The mask has one set bit for each trailing zero of the input.
 // Zero input produces an all-ones mask, whose population is 64.
 // Parallel sums count bits in pairs, nibbles, bytes, then words.
 uint64_t bits = ~in & (in - (uint64_t)1);
 bits -= (bits >> 1) & 0x5555555555555555ULL;
 bits = (bits & 0x3333333333333333ULL)
      + ((bits >> 2) & 0x3333333333333333ULL);
 bits = (bits + (bits >> 4)) & 0x0F0F0F0F0F0F0F0FULL;
 bits += bits >> 8;
 bits += bits >> 16;
 bits += bits >> 32;
 return (int)(bits & 0x7FULL);
}

static int ctz(uint64_t in)
{
 // Isolate the least significant set bit using unsigned negation.
 // Each mask tests one binary digit of that bit's position.
 // Their weighted sum is the trailing-zero count; zero returns 64.
 uint64_t bit;
 int r;
 if (in == 0)
  return 64;
 bit = in & ((uint64_t)0 - in);
 r = ((bit & 0xAAAAAAAAAAAAAAAAULL) != 0);
 r += 2 * ((bit & 0xCCCCCCCCCCCCCCCCULL) != 0);
 r += 4 * ((bit & 0xF0F0F0F0F0F0F0F0ULL) != 0);
 r += 8 * ((bit & 0xFF00FF00FF00FF00ULL) != 0);
 r += 16 * ((bit & 0xFFFF0000FFFF0000ULL) != 0);
 r += 32 * ((bit & 0xFFFFFFFF00000000ULL) != 0);
 return r;
}

static int ctz(uint64_t in)
{
 // Skip complete zero nibbles, then count remaining low zero bits.
 // Every skipped bit is zero and counting stops at the first one.
 // Return 64 explicitly for zero to avoid an endless nibble loop.
 int r = 0;
 if (in == 0)
  return 64;
 while ((in & 15) == 0)
 {
  r += 4;
  in >>= 4;
 }
 while ((in & 1) == 0)
 {
  r++;
  in >>= 1;
 }
 return r;
}

static int ctz(uint64_t in)
{
 // Test each nonempty low-bit prefix for being entirely zero.
 // Exactly the prefixes ending below the first set bit qualify.
 // For zero all 64 prefixes qualify; mask shifts are unsigned.
 uint64_t mask = 1;
 int i;
 int r = 0;
 for (i = 0; i < 64; i++)
 {
  r += ((in & mask) == 0);
  mask = (mask << 1) | (uint64_t)1;
 }
 return r;
}
