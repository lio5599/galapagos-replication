    uint64_t z = (uint64_t)a * b;
    uint64_t w = ((z * p0i) & UINT64_C(0x7FFFFFFF)) * p;
    uint64_t low = (z & UINT64_C(0xFFFFFFFF))
                 + (w & UINT64_C(0xFFFFFFFF));
    uint32_t high = (uint32_t)((z >> 32) + (w >> 32) + (low >> 32));
    uint32_t q = (uint32_t)(((uint64_t)high << 1)
                         | ((low & UINT64_C(0xFFFFFFFF)) >> 31));
    uint32_t d = q - p;
    return d + (p & (uint32_t)(0 - (d >> 31)));
}

    uint32_t q = (uint32_t)((z >> 31) + (w >> 31) + carry);
    uint32_t d = q - p;
    return d + ((d & UINT32_C(0x80000000)) ? p : 0);

static uint32_t modp_montymul(uint32_t a, uint32_t b, uint32_t p, uint32_t p0i) {
    uint64_t z = (uint64_t)a * b;
    uint64_t m = ((uint64_t)(uint32_t)z * p0i) & UINT64_C(0x7FFFFFFF);
    uint64_t sum = z + m * p;
    uint32_t d = (uint32_t)(sum >> 31) - p;
    uint32_t mask = (uint32_t)(0 - (d >> 31));
    uint32_t corrected = d + p;
    return (d & ~mask) | (corrected & mask);
}

static uint32_t modp_montymul(uint32_t a, uint32_t b, uint32_t p, uint32_t p0i) {
    uint64_t z = (uint64_t)a * b;
    uint64_t m = (z * p0i) & UINT64_C(0x7FFFFFFF);
    uint64_t sum = z + m * p;
    uint32_t q = (uint32_t)(sum >> 31);
    uint32_t d = q - p;
    uint32_t mask = (uint32_t)(0 - (d >> 31));
    return d ^ ((d ^ q) & mask);
}
