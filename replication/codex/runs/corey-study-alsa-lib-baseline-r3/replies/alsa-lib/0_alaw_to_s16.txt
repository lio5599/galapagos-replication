	case 3: t = (t + 256) * 4; break;
	case 4: t = (t + 256) * 8; break;
	case 5: t = (t + 256) * 16; break;
	case 6: t = (t + 256) * 32; break;

	int seg = (x >> 4) & 7;
	int sign = (x & 128) ? 1 : -1;
	int t = sign * ((int)(x & 15) * 16 + 8);

		t += sign * 256;

	}
