{
    // Delay byte truncation until after combining the terms.
    // Full-width complements have the same low eight bits as the reference's byte complements.
    // Higher bits cannot affect lower bits, including for arbitrary unsigned inputs.
    unsigned int a = (x - 26U) >> 8;
    unsigned int b = (x - 52U) >> 8;
    unsigned int c = (x - 62U) >> 8;
    unsigned int d = (0U - (x ^ 62U)) >> 8;

                     (~a & b & (x + ('a' - 26))) |

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Apply De Morgan's law to the OR of the reference's five masked terms.
    // Every term is byte-limited, so the complemented intersection yields the same byte.
    // All mask arithmetic remains unsigned, preserving wraparound for every input.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    unsigned int p = a & (x + 'A');
    unsigned int q = (~a & b) & (x + ('a' - 26));
    unsigned int r = (~b & c) & (x + ('0' - 52));
    unsigned int s = d & '-';
    unsigned int t = e & '_';

    return (int)(~(~p & ~q & ~r & ~s & ~t) & 255U);
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Store the five effective masks and candidate values in parallel arrays.
    // OR-reducing their intersections reproduces the five reference terms exactly.
    // Masks need not be all-zero or all-one, so out-of-range inputs are also preserved.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int masks[5];
    unsigned int values[5];
    unsigned int result = 0U;
    unsigned int i;

    masks[0] = a;
    masks[1] = (a ^ 255U) & b;
    masks[2] = (b ^ 255U) & c;
    masks[3] = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    masks[4] = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    values[0] = x + 'A';
    values[1] = x + ('a' - 26);
    values[2] = x + ('0' - 52);
    values[3] = '-';
    values[4] = '_';

    for (i = 0U; i < 5U; ++i)
        result |= masks[i] & values[i];

    return (int)result;
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Construct the result one bit at a time from the five effective masks.
    // Each bit is the OR of the same five mask-and-value bits used by the reference.
    // Examining all eight output bits also preserves partially set masks on unusual inputs.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U;
    unsigned int p = x + 'A';
    unsigned int q = x + ('a' - 26);
    unsigned int r = x + ('0' - 52);
    unsigned int result = 0U;
    unsigned int i;

    for (i = 0U; i < 8U; ++i) {
        unsigned int bit = ((a >> i) & (p >> i)) |
                           (((a ^ 255U) >> i) & (b >> i) & (q >> i)) |
                           (((b ^ 255U) >> i) & (c >> i) & (r >> i)) |
                           ((d >> i) & ((unsigned int)'-' >> i)) |
                           ((e >> i) & ((unsigned int)'_' >> i));
        result |= (bit & 1U) << i;
    }

    return (int)result;
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Accumulate the five terms using u OR v equals u XOR (v AND NOT u).
    // This adds only previously unset bits, even when multiple terms overlap.
    // The original unsigned masks preserve all boundary and out-of-range behavior.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int result = a & (x + 'A');
    unsigned int term;

    term = (a ^ 255U) & b & (x + ('a' - 26));
    result ^= term & ~result;
    term = (b ^ 255U) & c & (x + ('0' - 52));
    result ^= term & ~result;
    term = ((((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U) & '-';
    result ^= term & ~result;
    term = ((((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U) & '_';
    result ^= term & ~result;

    return (int)result;
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Treat the masked candidates as sets of output bits and test each bit for membership.
    // A bit is emitted exactly when at least one reference term contains it.
    // No assumption about valid Base64 inputs or mutually exclusive masks is needed.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int terms[5];
    unsigned int result = 0U;
    unsigned int bit;
    unsigned int i;

    terms[0] = a & (x + 'A');
    terms[1] = (a ^ 255U) & b & (x + ('a' - 26));
    terms[2] = (b ^ 255U) & c & (x + ('0' - 52));
    terms[3] = ((((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U) & '-';
    terms[4] = ((((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U) & '_';

    for (bit = 1U; bit < 256U; bit <<= 1) {
        for (i = 0U; i < 5U; ++i) {
            if ((terms[i] & bit) != 0U) {
                result |= bit;
                break;
            }
        }
    }

    return (int)result;
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Replace each eight-bit complement with subtraction from 255.
    // For any mask from 0 through 255, 255 minus the mask equals its XOR with 255.
    // This identity holds for partial masks too, preserving every unsigned input.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int d = ((0U - (x ^ 62U)) >> 8) & 255U;
    unsigned int e = ((0U - (x ^ 63U)) >> 8) & 255U;
    unsigned int result = (255U - e) & '_';

    result |= (255U - d) & '-';
    result |= (255U - b) & c & (x + ('0' - 52));
    result |= (255U - a) & b & (x + ('a' - 26));
    result |= a & (x + 'A');

    return (int)result;
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Extract each mask byte using division by 256 followed by remainder modulo 256.
    // For unsigned values these operations equal shifting right eight and masking with 255.
    // Subtractions occur in unsigned int first, retaining the reference's wraparound.
    unsigned int a = ((x - 26U) / 256U) % 256U;
    unsigned int b = ((x - 52U) / 256U) % 256U;
    unsigned int c = ((x - 62U) / 256U) % 256U;
    unsigned int d = (((0U - (x ^ 62U)) / 256U) % 256U) ^ 255U;
    unsigned int e = (((0U - (x ^ 63U)) / 256U) % 256U) ^ 255U;

    return (int)((a & (x + 'A')) |
                 ((a ^ 255U) & b & (x + ('a' - 26))) |
                 ((b ^ 255U) & c & (x + ('0' - 52))) |
                 (d & '-') |
                 (e & '_'));
}

static int b64_byte_to_urlsafe_char(unsigned int x)
{
    // Combine terms with the identity u OR v equals u plus v minus (u AND v).
    // Each operand is at most 255, so these additions cannot overflow unsigned int.
    // Overlapping and partial masks are handled exactly, including outside 0 through 63.
    unsigned int a = ((x - 26U) >> 8) & 255U;
    unsigned int b = ((x - 52U) >> 8) & 255U;
    unsigned int c = ((x - 62U) >> 8) & 255U;
    unsigned int result = a & (x + 'A');
    unsigned int term;

    term = (a ^ 255U) & b & (x + ('a' - 26));
    result = result + term - (result & term);
    term = (b ^ 255U) & c & (x + ('0' - 52));
    result = result + term - (result & term);
    term = ((((0U - (x ^ 62U)) >> 8) & 255U) ^ 255U) & '-';
    result = result + term - (result & term);
    term = ((((0U - (x ^ 63U)) >> 8) & 255U) ^ 255U) & '_';
    result = result + term - (result & term);

    return (int)result;
