{
    // The reference maps 0..63 to Base64 characters and otherwise computes a masked unsigned result.
    // Name each original mask and retain every unsigned operation.
    // Masks are not assumed to be Boolean, so out-of-range inputs also match.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU;
    return (int)((a & (x + 'A')) |
                 ((a ^ 0xFFU) & b & (x + ('a' - 26))) |
                 ((b ^ 0xFFU) & c & (x + ('0' - 52))) |
                 (d & '+') | (e & '/'));
}

static int b64_byte_to_char(unsigned int x)
{
    // Complement shifted words instead of XORing their extracted bytes.
    // Every complemented mask is restricted to eight bits before use.
    // This preserves the reference's wraparound and partial masks for all x.
    unsigned int a = (x - 26U) >> 8;
    unsigned int b = (x - 52U) >> 8;
    unsigned int c = (x - 62U) >> 8;
    unsigned int d = (0U - (x ^ 62U)) >> 8;
    unsigned int e = (0U - (x ^ 63U)) >> 8;
    return (int)(((a & (x + 'A')) |
                  (~a & b & (x + ('a' - 26))) |
                  (~b & c & (x + ('0' - 52))) |
                  (~d & '+') | (~e & '/')) & 0xFFU);
}

static int b64_byte_to_char(unsigned int x)
{
    // Store the five original contributions and OR them in a loop.
    // Each contribution uses the reference's exact unsigned masks.
    // No input-range assumption is made, including at unsigned wraparound.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int terms[5] = {
        a & (x + 'A'),
        (a ^ 0xFFU) & b & (x + ('a' - 26)),
        (b ^ 0xFFU) & (((x - 62U) >> 8) & 0xFFU) &
            (x + ('0' - 52)),
        ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+',
        ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/'
    };
    unsigned int result = 0U;
    unsigned int i;
    for (i = 0U; i < 5U; ++i)
        result |= terms[i];
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Pair each candidate value with its original selection mask.
    // The loop combines masked values rather than testing ranges.
    // Partial masks on out-of-range inputs therefore remain unchanged.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int masks[5] = {
        a,
        (a ^ 0xFFU) & b,
        (b ^ 0xFFU) & (((x - 62U) >> 8) & 0xFFU),
        ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU),
        ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU)
    };
    unsigned int values[5] = {
        x + 'A', x + ('a' - 26), x + ('0' - 52), '+', '/'
    };
    unsigned int result = 0U;
    unsigned int i;
    for (i = 0U; i < 5U; ++i)
        result |= masks[i] & values[i];
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Construct each of the eight result bits independently.
    // A bit is set exactly when an original contribution contains it.
    // This also handles overlapping or partial masks outside 0..63.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int result = 0U;
    unsigned int i;
    for (i = 0U; i < 8U; ++i) {
        unsigned int bit = 1U << i;
        if ((a & (x + 'A') & bit) ||
            ((a ^ 0xFFU) & b & (x + ('a' - 26)) & bit) ||
            ((b ^ 0xFFU) & c & (x + ('0' - 52)) & bit) ||
            (d & '+' & bit) || (e & '/' & bit))
            result |= bit;
    }
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Apply De Morgan's law to the union of the five contributions.
    // Complementing their intersection recovers precisely their bitwise OR.
    // Unsigned masks and arithmetic retain all out-of-range behavior.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int result = ~(a & (x + 'A'));
    result &= ~((a ^ 0xFFU) & b & (x + ('a' - 26)));
    result &= ~((b ^ 0xFFU) & c & (x + ('0' - 52)));
    result &= ~(d & '+');
    result &= ~(e & '/');
    return (int)~result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Extract candidate low bytes before applying the original masks.
    // Higher candidate bits cannot survive any eight-bit selection mask.
    // All additions remain unsigned, preserving wraparound for every x.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int upper = (x + 'A') & 0xFFU;
    unsigned int lower = (x + ('a' - 26)) & 0xFFU;
    unsigned int digit = (x + ('0' - 52)) & 0xFFU;
    unsigned int result = a & upper;
    result |= (0xFFU ^ a) & b & lower;
    result |= (0xFFU ^ b) & c & digit;
    result |= (0xFFU ^ (((0U - (x ^ 62U)) >> 8) & 0xFFU)) & '+';
    result |= (0xFFU ^ (((0U - (x ^ 63U)) >> 8) & 0xFFU)) & '/';
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Combine the first two contributions using a bitwise selection identity.
    // At each bit, a selects the uppercase value or the masked lowercase value.
    // This identity holds for partial masks, so arbitrary inputs still match.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int lower = b & (x + ('a' - 26));
    unsigned int result = lower ^ (a & (lower ^ (x + 'A')));
    result |= (b ^ 0xFFU) & c & (x + ('0' - 52));
    result |= ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+';
    result |= ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/';
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Accumulate contributions using XOR only for previously unset bits.
    // r XOR (t AND NOT r) equals r OR t, even when contributions overlap.
    // Original unsigned masks preserve all boundary and out-of-range cases.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int terms[5] = {
        a & (x + 'A'),
        (a ^ 0xFFU) & b & (x + ('a' - 26)),
        (b ^ 0xFFU) & (((x - 62U) >> 8) & 0xFFU) &
            (x + ('0' - 52)),
        ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+',
        ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/'
    };
    unsigned int result = 0U;
    unsigned int i;
    for (i = 0U; i < 5U; ++i)
        result ^= terms[i] & ~result;
    return (int)result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Replace unsigned shifts and byte masks with division and remainder.
    // For unsigned values, division by 256 and remainder modulo 256 are exact equivalents.
    // Subtracting an extracted byte from 255 complements its eight bits.
    // Subtractions occur before division, preserving unsigned wraparound for all x.
    unsigned int a = ((x - 26U) / 256U) % 256U;
    unsigned int b = ((x - 52U) / 256U) % 256U;

                 ((255U - a) & b & (x + ('a' - 26))) |
                 ((255U - b) & c & (x + ('0' - 52))) |
                 (d & '+') | (e & '/'));
