static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Computes the reference frame-size bound from header, sample, and CRC bytes.
    // A conditional expression selects the same stereo or general payload.
    // Signed division and addition order are unchanged, including zero or negatives.
    int count = 16;
    count += ch * ((7 + bps + 7) / 8);
    count += (ch == 2)
        ? ((2 * bps + 1) * blocksize + 7) / 8
        : (ch * bps * blocksize + 7) / 8;
    count += 2;
    return count;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // A switch selects exactly the same two channel cases.
    // All arithmetic and its grouping match the reference.
    // Zero and negative inputs retain the reference's signed division behavior.
    int count = 16;
    count += ch * ((7 + bps + 7) / 8);
    switch (ch) {
    case 2:
        count += ((2 * bps + 1) * blocksize + 7) / 8;
        break;
    default:
        count += (ch * bps * blocksize + 7) / 8;
        break;
    }
    count += 2;
    return count;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Reverses the branch order without changing which formula is selected.
    // The accumulator follows the original sequence of additions.
    // No positivity assumptions or division substitutions affect edge cases.
    int count = 16;
    count += ch * ((7 + bps + 7) / 8);
    if (ch != 2) {
        count += (ch * bps * blocksize + 7) / 8;
    } else {
        count += ((2 * bps + 1) * blocksize + 7) / 8;
    }
    count += 2;
    return count;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Stores the selected payload size before adding it to the accumulator.
    // Each expression and accumulator update preserves the original grouping.
    // Signed division remains identical for zero and negative operands.
    int count = 16;

    if (ch == 2) {

    } else {

    // Multiplication, bias, division, and accumulation keep their original order.
    // This also preserves signed truncation for nonpositive inputs.
    int count = 16;
    int bits;
    count += ch * ((7 + bps + 7) / 8);

        bits = ch * bps;

    count += 2;

    int count = 16;

    count += ch * ((7 + bps + 7) / 8);
    factor = ch == 2 ? 2 * bps + 1 : ch * bps;
    count += (factor * blocksize + 7) / 8;

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Breaks header arithmetic into steps matching its original evaluation.
    // The payload and accumulator additions are otherwise unchanged.
    // No reassociation or unsigned conversion changes overflow or negative cases.
    int count = 16;
    int header = 7 + bps;
    header += 7;
    header /= 8;
    header *= ch;
    count += header;
    if (ch == 2) {
        count += ((2 * bps + 1) * blocksize + 7) / 8;
    } else {
        count += (ch * bps * blocksize + 7) / 8;
    }
    count += 2;
    return count;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Computes payload bits first, then applies the original bias and division.
    // Each intermediate corresponds to an original signed-int operation.
    // Zero and negative values therefore follow the same truncation rules.
    int count = 16;
    int payload;
    count += ch * ((7 + bps + 7) / 8);
    if (ch == 2) {
        payload = 2 * bps;
        payload += 1;
    } else {
        payload = ch * bps;
    }
    payload *= blocksize;
    payload += 7;
    payload /= 8;
    count += payload;
    count += 2;
    return count;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Returns early for stereo after performing the original final addition.
    // Other channel counts take the general path with the same arithmetic.
    // Addition grouping and signed division are preserved even at edge values.
    int count = 16;
    count += ch * ((7 + bps + 7) / 8);
    if (ch == 2) {
        count += ((2 * bps + 1) * blocksize + 7) / 8;
        count += 2;
        return count;
    }
    count += (ch * bps * blocksize + 7) / 8;
    count += 2;
    return count;
}

static int flac_get_max_frame_size(int blocksize, int ch, int bps)
{
    // Stores successive accumulator states in separate local variables.
    // This preserves the original addition grouping rather than combining constants.
    // The same selected payload and signed division cover all defined edge cases.
    int initial = 16;
    int with_header = initial + ch * ((7 + bps + 7) / 8);
    int with_payload;
    int result;
    if (ch == 2) {
        with_payload = with_header + ((2 * bps + 1) * blocksize + 7) / 8;
    } else {
        with_payload = with_header + (ch * bps * blocksize + 7) / 8;
    }
    result = with_payload + 2;
    return result;
}
