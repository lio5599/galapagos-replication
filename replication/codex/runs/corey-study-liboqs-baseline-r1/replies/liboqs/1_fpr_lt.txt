static int fpr_lt(uint64_t x, uint64_t y) {
    if ((x >> 63) != (y >> 63)) {
        return (int)(x >> 63);
    }
    if (x >> 63) {
        return x > y;
    }
    return x < y;

    uint64_t ax = x & mask;
    uint64_t ay = y & mask;
    unsigned sx = (unsigned)(x >> 63);

    unsigned less = (unsigned)((ax - ay) >> 63);

        | ((1U ^ sx ^ sy) & ((less & (sx ^ 1U)) | (greater & sx))));

static int fpr_lt(uint64_t x, uint64_t y) {
    uint64_t select = (uint64_t)0 - (x >> 63);
    uint64_t a = (x & ~select) | (y & select);
    uint64_t b = (y & ~select) | (x & select);
    int different = (int)((x ^ y) >> 63);

    return (different & (int)(x >> 63))
        | ((different ^ 1) & (a < b));

static int fpr_lt(uint64_t x, uint64_t y) {
    uint32_t hx = (uint32_t)(x >> 32);
    uint32_t hy = (uint32_t)(y >> 32);
    uint32_t lx = (uint32_t)x;
    uint32_t ly = (uint32_t)y;

    if ((hx >> 31) != (hy >> 31)) {
        return (int)(hx >> 31);
    }
    if (hx >> 31) {
        return (hx > hy) || (hx == hy && lx > ly);
    }
    return (hx < hy) || (hx == hy && lx < ly);
}

/* Alternative 10 */
static int fpr_lt(uint64_t x, uint64_t y) {
    int sx = (int)(x >> 63);
    int sy = (int)(y >> 63);
    int order = (x > y) - (x < y);

    if (sx != sy) {
        return sx;
    }
    return order * (1 - 2 * sx) < 0;
}
