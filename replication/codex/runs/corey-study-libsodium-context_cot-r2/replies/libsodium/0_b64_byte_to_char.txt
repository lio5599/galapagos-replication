static int b64_byte_to_char(unsigned int x)
{
    // The reference computes a masked Base64 character mapping, including for inputs above 63.
    // Name each extracted byte mask without changing any unsigned arithmetic.
    // Preserve the masks exactly rather than assuming they are Boolean for arbitrary inputs.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int d = ((0U - (x ^ 62U)) >> 8) & 0xFFU;
    unsigned int e = ((0U - (x ^ 63U)) >> 8) & 0xFFU;
    return (a & (x + 'A')) |
           ((a ^ 0xFFU) & b & (x + ('a' - 26))) |
           ((b ^ 0xFFU) & c & (x + ('0' - 52))) |
           ((d ^ 0xFFU) & '+') |
           ((e ^ 0xFFU) & '/');
}

static int b64_byte_to_char(unsigned int x)
{
    // Accumulate the original five terms using associative bitwise OR.
    // Each subtraction remains unsigned, preserving wraparound for every input.
    // All extracted masks retain their complete byte values.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int result = a & (x + 'A');
    result |= (a ^ 0xFFU) & b & (x + ('a' - 26));
    result |= (b ^ 0xFFU) & c & (x + ('0' - 52));
    result |= ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+';
    result |= ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/';
    return result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Compute complemented byte masks by complementing before masking.
    // Complement and byte extraction commute, giving the original XOR masks.
    // Unsigned arithmetic and all five OR terms are preserved for edge inputs.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int na = (~((x - 26U) >> 8)) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int nb = (~((x - 52U) >> 8)) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int nd = (~((0U - (x ^ 62U)) >> 8)) & 0xFFU;
    unsigned int ne = (~((0U - (x ^ 63U)) >> 8)) & 0xFFU;
    return (a & (x + 'A')) |
           (na & b & (x + ('a' - 26))) |
           (nb & c & (x + ('0' - 52))) |
           (nd & '+') | (ne & '/');
}

static int b64_byte_to_char(unsigned int x)
{
    // Apply De Morgan's law to the final union of the five original terms.
    // This is a bitwise identity, including when extracted masks are not Boolean.
    // The intermediate terms retain the reference's unsigned wraparound behavior.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int t0 = a & (x + 'A');
    unsigned int t1 = (a ^ 0xFFU) & b & (x + ('a' - 26));
    unsigned int t2 = (b ^ 0xFFU) & c & (x + ('0' - 52));
    unsigned int t3 = ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+';
    unsigned int t4 = ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/';
    return ~(~t0 & ~t1 & ~t2 & ~t3 & ~t4);
}

static int b64_byte_to_char(unsigned int x)
{
    // Combine the first two terms with the standard bitwise selection identity.
    // The second candidate is byte-masked, so full complement and byte complement agree.
    // Selection acts independently on every bit, preserving arbitrary-input masks.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int p = x + 'A';
    unsigned int q = b & (x + ('a' - 26));
    unsigned int first = q ^ (a & (p ^ q));
    return first |
           ((b ^ 0xFFU) & c & (x + ('0' - 52))) |
           (((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+') |
           (((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/');
}

static int b64_byte_to_char(unsigned int x)
{
    // Remove excluded bits with XOR instead of intersecting complemented masks.
    // For byte-sized t, t XOR (m AND t) equals t AND the byte complement of m.
    // This identity holds for every mask value and preserves all unsigned edge cases.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int d = ((0U - (x ^ 62U)) >> 8) & 0xFFU;
    unsigned int e = ((0U - (x ^ 63U)) >> 8) & 0xFFU;
    unsigned int q = b & (x + ('a' - 26));
    unsigned int r = c & (x + ('0' - 52));
    unsigned int plus = ((unsigned int) '+') & 0xFFU;
    unsigned int slash = ((unsigned int) '/') & 0xFFU;
    return (a & (x + 'A')) |
           (q ^ (a & q)) |
           (r ^ (b & r)) |
           (plus ^ (d & plus)) |
           (slash ^ (e & slash));
}

static int b64_byte_to_char(unsigned int x)
{
    // Replace each final OR accumulation by u plus v minus their intersection.
    // That unsigned arithmetic identity equals bitwise OR for all bit patterns.
    // The original masks and terms remain unchanged, including outside Base64's range.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int result = a & (x + 'A');
    unsigned int term = (a ^ 0xFFU) & b & (x + ('a' - 26));
    result = result + term - (result & term);
    term = (b ^ 0xFFU) & c & (x + ('0' - 52));
    result = result + term - (result & term);
    term = ((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+';
    result = result + term - (result & term);
    term = ((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/';
    result = result + term - (result & term);
    return result;
}

static int b64_byte_to_char(unsigned int x)
{
    // Evaluate the low and high nibbles separately, then join them.
    // Every reference term is byte-masked, and AND distributes over OR.
    // The nibble partition therefore loses no bits, even for unrestricted inputs.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int d = (((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int e = (((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU;
    unsigned int p = x + 'A';
    unsigned int q = x + ('a' - 26);
    unsigned int r = x + ('0' - 52);
    unsigned int low = (a & 0x0FU & p) |
                       ((a ^ 0xFFU) & b & 0x0FU & q) |
                       ((b ^ 0xFFU) & c & 0x0FU & r) |
                       (d & 0x0FU & '+') | (e & 0x0FU & '/');
    unsigned int high = (a & 0xF0U & p) |
                        ((a ^ 0xFFU) & b & 0xF0U & q) |
                        ((b ^ 0xFFU) & c & 0xF0U & r) |
                        (d & 0xF0U & '+') | (e & 0xF0U & '/');
    return low | high;
}

static int b64_byte_to_char(unsigned int x)
{
    // Postpone byte masking until after combining all terms.
    // Only the low eight bits can contribute in the reference.
    // Full complements have identical low bits to the reference's byte complements.
    // All arithmetic before extraction remains unsigned for wraparound inputs.
    unsigned int a = (x - 26U) >> 8;
    unsigned int b = (x - 52U) >> 8;
    unsigned int c = (x - 62U) >> 8;
    unsigned int d = (0U - (x ^ 62U)) >> 8;
    unsigned int e = (0U - (x ^ 63U)) >> 8;
    unsigned int result = (a & (x + 'A')) |
                          (~a & b & (x + ('a' - 26))) |
                          (~b & c & (x + ('0' - 52))) |
                          (~d & '+') |
                          (~e & '/');
    return result & 0xFFU;
}

static int b64_byte_to_char(unsigned int x)
{
    // Replace the first OR by XOR because its two terms have disjoint masks.
    // One term uses a and the other its byte complement, so they cannot overlap.
    // Keep OR for the remaining terms, whose masks may overlap on arbitrary inputs.
    unsigned int a = ((x - 26U) >> 8) & 0xFFU;
    unsigned int b = ((x - 52U) >> 8) & 0xFFU;
    unsigned int c = ((x - 62U) >> 8) & 0xFFU;
    unsigned int first = (a & (x + 'A')) ^
                         ((a ^ 0xFFU) & b & (x + ('a' - 26)));
    unsigned int digits = (b ^ 0xFFU) & c & (x + ('0' - 52));
    unsigned int punctuation =
        (((((0U - (x ^ 62U)) >> 8) & 0xFFU) ^ 0xFFU) & '+') |
        (((((0U - (x ^ 63U)) >> 8) & 0xFFU) ^ 0xFFU) & '/');
    return first | digits | punctuation;
}
