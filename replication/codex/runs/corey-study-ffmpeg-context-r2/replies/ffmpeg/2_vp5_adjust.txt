static int vp5_adjust(int v, int t)
{
    int sign = v >> 31;
    int a = (v ^ sign) - sign;
    int d = (a < 2 * t ? a : 0) - t;
    int mask = d >> 31;
    int r = t - ((d ^ mask) - mask);
    return (r + sign) ^ sign;
}

static int vp5_adjust(int v, int t)
{
    int negative = v < 0;
    int a = negative ? -v : v;
    int d;

    if (a >= 2 * t)
        a = 0;

    d = a - t;
    if (d < 0)
        d = -d;

    a = t - d;
    return negative ? -a : a;
}
