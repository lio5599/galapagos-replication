static uint64_t icbrt64(uint64_t x)
{
    uint64_t r = 0;
    int s = 63;

    do {
        r <<= 1;
        uint64_t b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            ++r;
        }
        s -= 3;
    } while (s >= 0);

    return r << 12;
}

{
    uint64_t r = 0;
    uint64_t remainder = 0;

    for (int s = 63; s >= 0; s -= 3) {
        remainder = (remainder << 3) | ((x >> s) & 7);
        r <<= 1;

        uint64_t delta = 3 * r * (r + 1) + 1;
        if (remainder >= delta) {
            remainder -= delta;
            ++r;
        }
    }

    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    if (x == 0)
        return 0;

    uint64_t r = (uint64_t)1 << 22;

    for (;;) {
        uint64_t next = (2 * r + x / r / r) / 3;

        if (next >= r)
            return r << 12;

        r = next;
    }
}

static uint64_t icbrt64(uint64_t x)
{
    uint64_t r = 0;

    for (uint64_t place = 1000000; place; place /= 10) {
        for (unsigned digit = 9; digit != 0; --digit) {
            uint64_t candidate = r + digit * place;

            if (candidate <= x / candidate / candidate) {
                r = candidate;
                break;
            }
        }
    }

    return r << 12;
}
