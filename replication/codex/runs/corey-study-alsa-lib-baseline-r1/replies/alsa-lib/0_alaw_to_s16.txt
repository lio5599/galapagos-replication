{

    a_val ^= 0x55;
    segment = (a_val >> 4) & 7;
    magnitude = (a_val & 15) * 16 + 8;

    return (a_val & 0x80) ? magnitude : -magnitude;

static int alaw_to_s16(unsigned char a_val)
{
    int decoded = a_val ^ 0x55;
    int segment = (decoded >> 4) & 7;
    int magnitude = (decoded & 15) * 16 + 8;
    int shift;

    if (segment == 0)
        return (decoded & 128) ? magnitude : -magnitude;

    magnitude += 256;
    shift = segment - 1;

    if (shift & 1)
        magnitude <<= 1;
    if (shift & 2)
        magnitude <<= 2;
    if (shift & 4)
        magnitude <<= 4;

    return (decoded & 128) ? magnitude : -magnitude;
}

static int alaw_to_s16(unsigned char a_val)
{
    int decoded = a_val ^ 0x55;
    int segment = (decoded >> 4) & 7;
    int mantissa = decoded & 15;
    int magnitude;

    if (segment == 0) {
        magnitude = mantissa * 16 + 8;
    } else {
        int scale = 1 << (segment + 3);
        magnitude = mantissa * scale + scale / 2 + (1 << (segment + 7));
    }

    if (!(decoded & 128))
        magnitude = -magnitude;

    return magnitude;
}
