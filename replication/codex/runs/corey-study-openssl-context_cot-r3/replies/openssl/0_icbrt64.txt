static uint64_t icbrt64(uint64_t x)
{
    // Computes floor(cube_root(x)) multiplied by 4096.
    // Expanding the trial polynomial preserves each reference decision.
    // All intermediate polynomial values fit, including at UINT64_MAX.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        r <<= 1;
        uint64_t b = 3 * r * r + 3 * r + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            ++r;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Counts the same 22 rounds using an unsigned iteration counter.
    // The shifts remain 63, 60, through 0, so every decision is unchanged.
    // Zero and the largest input follow exactly the reference recurrence.
    uint64_t r = 0;
    for (unsigned k = 22; k != 0; ) {
        --k;
        unsigned s = 3 * k;
        r += r;
        uint64_t b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            r += 1;
        }
    }
    return r * 4096;
}

static uint64_t icbrt64(uint64_t x)
{
    // A do-while loop performs the same rounds and stops after shift zero.
    // Subtracting three happens only when another round remains.
    // The arithmetic and decisions match the reference for every input.
    uint64_t r = 0;
    int s = 63;
    do {
        r <<= 1;
        uint64_t b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            ++r;
        }
        if (s == 0)
            break;
        s -= 3;
    } while (1);
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Replaces the conditional update with an unsigned all-or-zero mask.
    // A failed trial subtracts zero and a successful trial subtracts b << s.
    // Unsigned mask arithmetic is defined even for zero and maximal inputs.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        r <<= 1;
        uint64_t b = 3 * r * (r + 1) + 1;
        uint64_t take = (uint64_t)((x >> s) >= b);
        uint64_t mask = (uint64_t)0 - take;
        x -= (b << s) & mask;
        r += take;
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Forms three times r times r plus one using additions.
    // The product and its triple fit in uint64_t throughout all 22 rounds.
    // Thus the trial thresholds and edge-case behavior are unchanged.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        r += r;
        uint64_t p = r * (r + 1);
        uint64_t b = p + p + p + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            ++r;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Computes the trial threshold before doubling the partial root.
    // Substituting twice r gives 12*r*r + 6*r + 1.
    // These values fit, preserving every trial even at the input extremes.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        uint64_t b = 12 * r * r + 6 * r + 1;
        r <<= 1;
        if ((x >> s) >= b) {
            x -= b << s;
            ++r;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Updates a successful remainder by splitting its high and low bits.
    // Subtracting b from the high part equals subtracting b shifted by s.
    // Shifts stay within 0 through 63; shift zero gives an empty low part.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        r <<= 1;
        uint64_t b = 3 * r * (r + 1) + 1;
        uint64_t high = x >> s;
        if (high >= b) {
            uint64_t low_mask = ((uint64_t)1 << s) - 1;
            x = ((high - b) << s) | (x & low_mask);
            ++r;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Rejects unsuccessful trials early instead of nesting successful ones.
    // The complementary comparison selects exactly the reference updates.
    // Every input still executes all 22 rounds, including zero.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        r <<= 1;
        uint64_t b = 3 * r * (r + 1) + 1;
        if ((x >> s) < b)
            continue;
        x -= b << s;
        ++r;
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Unrolls two successive reference rounds into each loop iteration.
    // The paired shifts run from 63 and 60 down to 3 and 0.
    // Their order and remainder updates are identical for all inputs.
    uint64_t r = 0;
    for (int s = 63; s >= 3; s -= 6) {
        r <<= 1;
        uint64_t b = 3 * r * (r + 1) + 1;
        if ((x >> s) >= b) {
            x -= b << s;
            ++r;
        }

        r <<= 1;
        b = 3 * r * (r + 1) + 1;
        if ((x >> (s - 3)) >= b) {
            x -= b << (s - 3);
            ++r;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Keeps the doubled root separate until the trial has been decided.
    // The next root is twice the old root plus the successful-trial bit.
    // Identical thresholds and remainder updates cover all input values.
    uint64_t r = 0;
    for (int s = 63; s >= 0; s -= 3) {
        uint64_t doubled = r << 1;
        uint64_t b = 3 * doubled * (doubled + 1) + 1;
        uint64_t take = (uint64_t)((x >> s) >= b);
        if (take != 0)
            x -= b << s;
        r = doubled | take;
    }
    return r << 12;
}
