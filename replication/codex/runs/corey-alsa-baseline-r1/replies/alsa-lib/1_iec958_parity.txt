static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	int bit;

	data >>= 4;
	for (bit = 0; bit < 27; ++bit) {
		parity ^= data & 1U;
		data >>= 1;
	}
	return parity;
}

static unsigned int iec958_parity(unsigned int data)
{
	unsigned int parity = 0;
	unsigned int mask = 0x10U;
	int bit;

	for (bit = 4; bit <= 30; ++bit) {
		parity ^= (data & mask) != 0U;
		mask <<= 1;
	}
	return parity;
}

	for (group = 0; group < 9; ++group) {

{
	data = (data >> 4) & 0x07ffffffU;
	data -= (data >> 1) & 0x55555555U;
	data = (data & 0x33333333U) +
	       ((data >> 2) & 0x33333333U);

	data += data >> 8;
	data += data >> 16;

	return data & 1U;
