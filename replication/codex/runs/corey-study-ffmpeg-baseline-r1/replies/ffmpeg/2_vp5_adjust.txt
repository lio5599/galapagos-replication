static int vp5_adjust(int v, int t)
{
    int negative = v < 0;
    int a = negative ? -v : v;
    int d, r;

    if (a >= 2 * t)
        a = 0;
    d = a - t;
    r = t - (d < 0 ? -d : d);
    return negative ? -r : r;
}

static int vp5_adjust(int v, int t)
{
    int sign = v >> 31;
    int a = (v ^ sign) - sign;
    int d = (a < 2 * t ? a : 0) - t;
    int mask = d >> 31;
    int r = t - ((d ^ mask) - mask);

    return (r + sign) ^ sign;
}

static int vp5_adjust(int v, int t)
{
    int negative = v < 0;
    int a = negative ? -v : v;
    int d, r;

    switch (a < 2 * t) {
    case 0:
        d = -t;
        break;
    default:
        d = a - t;
        break;
    }

    if (d < 0)
        d = -d;
    r = t - d;
    return negative ? -r : r;
}

static int vp5_adjust(int v, int t)
{
    int sign = v >> 31;
    int a = (v ^ sign) - sign;
    int d, mask, r;

    a &= -(a < 2 * t);
    d = a - t;

    return (r + sign) ^ sign;
}
