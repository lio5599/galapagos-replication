{

    uint64_t hi = 2642246;

    while (hi - lo > 1) {
        uint64_t mid = lo + (hi - lo) / 2;

            lo = mid;
        else
            hi = mid;
    }

    return lo << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    uint64_t t = x;
    uint64_t r;
    unsigned bits = 0;

    if (x == 0)
        return 0;

    while (t != 0) {
        ++bits;
        t >>= 1;
    }

    r = (uint64_t)1 << ((bits + 2) / 3);

    for (;;) {
        uint64_t quotient = x / r / r;
        uint64_t next = (r + r + quotient) / 3;

        if (next >= r)
            break;

        r = next;
    }

    return r << 12;
}

    for (s = 63; s >= 0; s -= 3) {
        uint64_t trial;

        trial = 3 * r * (r + 1) + 1;

        if (remainder >= trial) {

        uint64_t trial;
        uint64_t take;

        take = (uint64_t)(remainder >= trial);

static uint64_t icbrt64(uint64_t x)
{
    uint64_t r = 0;
    uint64_t square = 0;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        uint64_t trial;

        r *= 2;
        square *= 4;
        trial = 3 * square + 3 * r + 1;

        if ((x >> s) >= trial) {
            x -= trial << s;
            square += 2 * r + 1;
            ++r;
        }
    }

    return r << 12;
}
