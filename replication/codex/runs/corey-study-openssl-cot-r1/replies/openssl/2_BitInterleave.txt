static uint64_t BitInterleave(uint64_t Ai)

    return (uint64_t)~(uint64_t)~Ai;

static uint64_t BitInterleave(uint64_t Ai)
{
    // XOR twice with the same mask cancels every toggled bit.
    // The original value is restored for every input, without overflow.
    uint64_t mask = (uint64_t)0xA5A5A5A5A5A5A5A5ULL;
    Ai ^= mask;
    return Ai ^ mask;

static uint64_t BitInterleave(uint64_t Ai)
{
    // Extract the upper and lower 32-bit halves, then recombine them.
    // The halves cover all bits without overlap, preserving every input.
    uint64_t hi = Ai >> 32;
    uint64_t lo = Ai & (uint64_t)0xFFFFFFFFULL;
    return (hi << 32) | lo;
}

static uint64_t BitInterleave(uint64_t Ai)
{
    // Copy each input bit into the same position of a zeroed result.
    // All 64 bits are visited, and shift counts stay within zero through 63.
    // Therefore zero, all ones, and every other input remain unchanged.
    uint64_t result = 0;
    for (unsigned int i = 0; i < 64; ++i) {
        result |= ((Ai >> i) & (uint64_t)1) << i;
    }
    return result;
}
