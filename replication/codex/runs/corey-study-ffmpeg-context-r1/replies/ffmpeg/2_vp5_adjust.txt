static int vp5_adjust(int v, int t)
{
    int s2, s1 = v >> 31;
    v ^= s1;
    v -= s1;
    if (!(v < 2 * t))
        v = 0;
    v -= t;
    s2 = v >> 31;
    v ^= s2;
    v -= s2;
    v = t - v;
    v += s1;
    v ^= s1;
    return v;
}

    int sign = v >> 31;
    int magnitude = (v + sign) ^ sign;
    int difference;
    int difference_sign;
    int result;

    magnitude *= magnitude < 2 * t;
    difference = magnitude - t;
    difference_sign = difference >> 31;
    difference = (difference + difference_sign) ^ difference_sign;
    result = t - difference;

    return (result ^ sign) - sign;
}

static int vp5_adjust(int v, int t)
{
    int sign = v >> 31;
    int magnitude = (v ^ sign) - sign;
    int twice = 2 * t;

    if (magnitude < twice) {
        if (magnitude < t)
            return (magnitude + sign) ^ sign;
        return ((twice - magnitude) + sign) ^ sign;
    }

    if (t < 0)
        return (twice + sign) ^ sign;

    return 0;
}
