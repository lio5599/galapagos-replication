static int mix(int c0, int c1)
{
    int blue = (c0 & 0x001F) * 2 + (c1 & 0x001F);
    int green = ((c0 >> 5) & 31) * 2 + ((c1 >> 5) & 31);
    int red = (c0 >> 10) * 2 + (c1 >> 10);
    return red / 3 * 1024 + green / 3 * 32 + blue / 3;
}

static int mix(int c0, int c1)
{
    int blue = 2 * (c0 & 31) + (c1 & 31);
    int green = (2 * (c0 & 0x03E0) + (c1 & 0x03E0)) >> 5;
    int red = 2 * (c0 >> 10) + (c1 >> 10);
    int blue_q = (blue - blue % 3) / 3;
    int green_q = (green - green % 3) / 3;
    int red_q = (red - red % 3) / 3;
    return red_q * 1024 + green_q * 32 + blue_q;
}

static int mix(int c0, int c1)
{
    int blue = (c0 & 31);
    int green = (c0 & 0x03E0);
    int red = c0 >> 10;
    blue *= 2;
    green *= 2;
    red *= 2;
    blue += c1 & 31;
    green += c1 & 0x03E0;
    red += c1 >> 10;
    blue /= 3;
    green = (green >> 5) / 3;
    red /= 3;
    red *= 1024;
    green *= 32;
    return red + green + blue;
}
