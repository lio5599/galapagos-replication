static int vp5_adjust(int v, int t)
{
    int negative = v < 0;
    int magnitude = negative ? -v : v;
    int delta, result;

    if (magnitude >= 2 * t)
        magnitude = 0;

    delta = magnitude - t;
    result = t - (delta < 0 ? -delta : delta);
    return negative ? -result : result;
}

    int delta, mask, result;

static int vp5_adjust(int v, int t)
{
    int negative = v < 0;
    int magnitude = negative ? -v : v;
    int result;

    if (magnitude >= 2 * t)
        goto outside;
    if (magnitude > t)
        goto descending;

    result = magnitude;
    goto finish;

descending:
    result = 2 * t - magnitude;
    goto finish;

outside:
    result = t < 0 ? 2 * t : 0;

finish:
    return negative ? -result : result;
}

static int vp5_adjust(int v, int t)
{
    int negative = v < 0;
    int magnitude = negative ? -v : v;
    int result;

    switch (magnitude < 2 * t) {
    case 0:
        result = t - (t < 0 ? -t : t);
        break;
    default:
        result = magnitude <= t ? magnitude : 2 * t - magnitude;
        break;
    }

    return negative ? -result : result;
}
