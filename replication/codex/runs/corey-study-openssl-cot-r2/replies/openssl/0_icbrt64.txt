static uint64_t icbrt64(uint64_t x)
{
    // The reference computes floor(cube_root(x)) multiplied by 4096.
    // Binary search maintains a feasible lower and infeasible upper bound.
    // Division tests cubes without overflow; every tested midpoint is positive.
    // The bounds cover all uint64_t inputs, including zero and UINT64_MAX.
    uint64_t lo = 0;
    uint64_t hi = 2642246;

    while (hi - lo > 1) {
        uint64_t m = lo + (hi - lo) / 2;
        if (m <= x / m / m)
            lo = m;
        else
            hi = m;
    }
    return lo << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Search the inclusive range of possible uint64_t cube roots.
    // Every candidate is at most 2642245, so its cube cannot overflow.
    // Save the largest candidate whose cube does not exceed x.
    // Zero and the maximum input are covered; scaling matches the reference.
    uint64_t lo = 0;
    uint64_t hi = 2642245;
    uint64_t r = 0;

    while (lo <= hi) {
        uint64_t m = lo + (hi - lo) / 2;
        uint64_t cube = m * m * m;
        if (cube <= x) {
            r = m;
            lo = m + 1;
        } else {
            hi = m - 1;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Construct the root greedily from its highest possible bit downward.
    // A bit is retained exactly when the candidate cube does not exceed x.
    // Positive-candidate division avoids overflow even for rejected roots.
    // All 22 root bits are tested, handling zero and UINT64_MAX exactly.
    uint64_t r = 0;
    uint64_t bit;

    for (bit = (uint64_t)1 << 21; bit != 0; bit >>= 1) {
        uint64_t candidate = r | bit;
        if (candidate <= x / candidate / candidate)
            r = candidate;
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Consume three input bits at a time, retaining a cube remainder.
    // After doubling the root, the next cube costs 3*r*(r+1)+1.
    // Subtracting that cost when possible chooses the next exact root bit.
    // All intermediate values fit uint64_t, including at the input extremes.
    uint64_t r = 0;
    uint64_t remainder = 0;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        uint64_t cost;
        remainder = (remainder << 3) | ((x >> s) & 7);
        r <<= 1;
        cost = 3 * r * (r + 1) + 1;
        if (remainder >= cost) {
            remainder -= cost;
            ++r;
        }
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Integer Newton iteration starts above every possible cube root.
    // Each decreasing step stays at or above the floor of the true root.
    // A nondecreasing step therefore identifies that floor exactly.
    // Zero is handled separately; division and the bounded sum cannot overflow.
    uint64_t r = (uint64_t)1 << 22;

    if (x == 0)
        return 0;

    for (;;) {
        uint64_t next = (2 * r + x / r / r) / 3;
        if (next >= r)
            return r << 12;
        r = next;
    }
}

static uint64_t icbrt64(uint64_t x)
{
    // Increment the root while subtracting consecutive cube differences.
    // The residual always equals the original input minus r cubed.
    // Failure to afford the next difference means r is the floor cube root.
    // Differences fit uint64_t, and guarded subtraction handles both extremes.
    uint64_t r = 0;

    for (;;) {
        uint64_t difference = 3 * r * (r + 1) + 1;
        if (x < difference)
            break;
        x -= difference;
        ++r;
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Generate consecutive cubes using first and second finite differences.
    // The first difference starts at 1; its increments are 6, 12, 18, and so on.
    // Guard each addition with the remaining input to prevent cube overflow.
    // The last affordable cube gives the exact root, including at both extremes.
    uint64_t r = 0;
    uint64_t cube = 0;
    uint64_t difference = 1;
    uint64_t increment = 6;

    while (x - cube >= difference) {
        cube += difference;
        ++r;
        difference += increment;
        increment += 6;
    }
    return r << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Narrow the root interval using two cube probes per iteration.
    // Monotonic cubes preserve an interval containing the exact floor root.
    // Candidates never exceed 2642245, so both cube products fit uint64_t.
    // The interval strictly shrinks, including for zero and the maximum input.
    uint64_t lo = 0;
    uint64_t hi = 2642245;

    while (lo < hi) {
        uint64_t third = (hi - lo) / 3;
        uint64_t a = lo + third;
        uint64_t b = hi - third;

        if (a * a * a > x) {
            hi = a - 1;
        } else if (b * b * b <= x) {
            lo = b;
        } else {
            lo = a;
            hi = b - 1;
        }
    }
    return lo << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Exponential search brackets the root, then binary search finishes.
    // Division gives exact cube comparisons without overflowing products.
    // The upper bound reaches at most 4194304, so doubling is also safe.
    // Zero returns directly; all positive uint64_t inputs share the same search.
    uint64_t lo = 0;
    uint64_t hi = 1;

    if (x == 0)
        return 0;

    while (hi <= x / hi / hi) {
        lo = hi;
        hi <<= 1;
    }

    while (hi - lo > 1) {
        uint64_t m = lo + (hi - lo) / 2;
        if (m <= x / m / m)
            lo = m;
        else
            hi = m;
    }
    return lo << 12;
}

static uint64_t icbrt64(uint64_t x)
{
    // Use the reference's restoring steps while maintaining the root square.
    // Doubling multiplies the square by four; incrementing adds 2*r+1.
    // Thus 3*(square+r)+1 is exactly the reference's trial difference.
    // Its comparison guards the shifted subtraction, including at both extremes.
    uint64_t r = 0;
    uint64_t square = 0;
    int s;

    for (s = 63; s >= 0; s -= 3) {
        uint64_t difference;
        r <<= 1;
        square <<= 2;
        difference = 3 * (square + r) + 1;

        if ((x >> s) >= difference) {
            x -= difference << s;
            square += 2 * r + 1;
            ++r;
        }
    }
    return r << 12;
}
