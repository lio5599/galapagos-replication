    unsigned int b = (x - 52U) >> 8;
    unsigned int c = (x - 62U) >> 8;
    unsigned int lower = b & (x + ('a' - 26));
    unsigned int letters = (a | lower) & (~a | (x + 'A'));
    unsigned int digits = ~b & c & (x + ('0' - 52));
    unsigned int symbols =
        (~((0U - (x ^ 62U)) >> 8) & '+') |
        (~((0U - (x ^ 63U)) >> 8) & '/');

    return (letters | digits | symbols) & 255U;
}
